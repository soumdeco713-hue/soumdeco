     1→"use client";
     2→
     3→import { useEffect, useMemo, useState } from "react";
     4→import { toast } from "sonner";
     5→import { Check, Send, Heart, Sparkles } from "lucide-react";
     6→import {
     7→  getShippingPrice,
     8→  ShippingCompany,
     9→  DeliveryType,
    10→  SHIPPING_COMPANY_LABELS_AR,
    11→  DELIVERY_TYPE_LABELS_AR,
    12→} from "@/lib/shipping";
    13→import { formatPrice, QuantityTier } from "@/lib/products";
    14→import { useAlgeriaData, PHONE_REGEX } from "@/hooks/use-algeria-data";
    15→import { BRAND } from "@/lib/brand-config";
    16→
    17→export type OrderItem = {
    18→  name: string;
    19→  price: number | null;
    20→  quantity: number;
    21→};
    22→
    23→type CodOrderFormProps = {
    24→  items: OrderItem[];
    25→  rupture?: boolean;
    26→  onSuccess?: () => void;
    27→  onContinueShopping?: () => void;
    28→  /** Extra notes that should be appended to the order notes (e.g. variation selections). */
    29→  extraNotes?: string;
    30→  /** Optional quantity tiers — when provided AND items.length === 1, the 4-button
    31→   *  selector highlights tier quantities with a ✨ badge and applies discount /
    32→   *  free-shipping benefits to the totals. */
    33→  quantityTiers?: QuantityTier[];
    34→};
    35→
    36→type FormState = {
    37→  fullName: string;
    38→  phone: string;
    39→  wilaya: string;
    40→  commune: string;
    41→  company: ShippingCompany;
    42→  delivery: DeliveryType;
    43→  notes: string;
    44→};
    45→
    46→function generateOrderRef(): string {
    47→  const n = Math.floor(100000 + Math.random() * 900000);
    48→  return `SD-${n}`;
    49→}
    50→
    51→export function CodOrderForm({
    52→  items: initialItems,
    53→  rupture,
    54→  onSuccess,
    55→  onContinueShopping,
    56→  extraNotes,
    57→  quantityTiers,
    58→}: CodOrderFormProps) {
    59→  const { wilayas, getCommunesForWilaya, loading } = useAlgeriaData();
    60→  const [items, setItems] = useState<OrderItem[]>(initialItems);
    61→  const [form, setForm] = useState<FormState>({
    62→    fullName: "",
    63→    phone: "",
    64→    wilaya: "",
    65→    commune: "",
    66→    company: "zr_express",
    67→    delivery: "stop_desk",
    68→    notes: "",
    69→  });
    70→  const [phoneError, setPhoneError] = useState(false);
    71→  const [submitting, setSubmitting] = useState(false);
    72→  const [done, setDone] = useState(false);
    73→  const [orderRef, setOrderRef] = useState("");
    74→  const [customQty, setCustomQty] = useState(""); // local input text — independent from buttons
    75→  const [orderSummary, setOrderSummary] = useState<null | {
    76→    items: OrderItem[];
    77→    productTotal: number;
    78→    discountAmount: number;
    79→    shippingPrice: number;
    80→    grandTotal: number;
    81→    freeShippingApplied: boolean;
    82→    fullName: string;
    83→    phone: string;
    84→    wilayaLabel: string;
    85→    deliveryLabel: string;
    86→    date: string;
    87→  }>(null);
    88→
    89→  // Sync items when prop changes (e.g., modal opens different product)
    90→  useEffect(() => {
    91→    setItems(initialItems);
    92→  }, [initialItems]);
    93→
    94→  const communes = useMemo(
    95→    () => (form.wilaya ? getCommunesForWilaya(form.wilaya) : []),
    96→    [form.wilaya, getCommunesForWilaya],
    97→  );
    98→
    99→  // Reset commune when wilaya changes
   100→  useEffect(() => {
   101→    setForm((f) => ({ ...f, commune: "" }));
   102→  }, [form.wilaya]);
   103→
   104→  // The NORMAL shipping price (without any tier benefit applied).
   105→  // Returns null until BOTH wilaya AND commune are selected.
   106→  const normalShippingPrice = useMemo(() => {
   107→    if (!form.wilaya || !form.commune) return null;
   108→    const result = getShippingPrice(form.wilaya, form.company, form.delivery);
   109→    return result ? result.price : null;
   110→  }, [form.wilaya, form.commune, form.company, form.delivery]);
   111→
   112→  // Active tier — only for single-item orders where the parent passed `quantityTiers`.
   113→  // Used to compute discount and free-shipping benefits.
   114→  const activeTier = useMemo(() => {
   115→    if (items.length !== 1) return null;
   116→    const q = items[0].quantity;
   117→    return (quantityTiers ?? []).find((t) => t.qty === q) ?? null;
   118→  }, [items, quantityTiers]);
   119→
   120→  // Discount from the active tier (if any) — clamped so total can never go negative.
   121→  const discountAmount = activeTier?.discountAmount ?? 0;
   122→
   123→  // Whether the active tier grants free shipping for the current delivery type.
   124→  const freeShippingApplied = useMemo(() => {
   125→    if (!activeTier || activeTier.freeShipping === "none") return false;
   126→    if (activeTier.freeShipping === "both") return true;
   127→    if (activeTier.freeShipping === "desk" && form.delivery === "stop_desk") return true;
   128→    if (activeTier.freeShipping === "home" && form.delivery === "home") return true;
   129→    return false;
   130→  }, [activeTier, form.delivery]);
   131→
   132→  const productTotal = useMemo(
   133→    () => items.reduce((s, i) => s + (i.price ?? 0) * i.quantity, 0),
   134→    [items],
   135→  );
   136→
   137→  // Product total AFTER discount (never negative).
   138→  const productTotalAfterDiscount = useMemo(
   139→    () => Math.max(0, productTotal - discountAmount),
   140→    [productTotal, discountAmount],
   141→  );
   142→
   143→  // Final shipping price: 0 if the tier grants free shipping, otherwise normal price.
   144→  const shippingPrice = useMemo(() => {
   145→    if (normalShippingPrice === null) return null;
   146→    return freeShippingApplied ? 0 : normalShippingPrice;
   147→  }, [normalShippingPrice, freeShippingApplied]);
   148→
   149→  const grandTotal = useMemo(() => {
   150→    const ship = shippingPrice ?? 0;
   151→    return productTotalAfterDiscount + ship;
   152→  }, [productTotalAfterDiscount, shippingPrice]);
   153→
   154→  const set = <K extends keyof FormState>(key: K, value: FormState[K]) => {
   155→    setForm((f) => ({ ...f, [key]: value }));
   156→    if (key === "phone") setPhoneError(false);
   157→  };
   158→
   159→  const setSingleQty = (q: number) => {
   160→    if (q < 1) return;
   161→    setItems((prev) =>
   162→      prev.length === 1 ? [{ ...prev[0], quantity: q }] : prev,
   163→    );
   164→  };
   165→
   166→  const tierBenefitLine = (tier: QuantityTier): string => {
   167→    const hasFree = tier.freeShipping && tier.freeShipping !== "none";
   168→    const discount = tier.discountAmount ?? 0;
   169→    const hasDiscount = discount > 0;
   170→    const shippingLabel =
   171→      tier.freeShipping === "both"
   172→        ? "كلاهما"
   173→        : tier.freeShipping === "desk"
   174→        ? "المكتب"
   175→        : tier.freeShipping === "home"
   176→        ? "المنزل"
   177→        : "";
   178→    if (hasFree && hasDiscount) {
   179→      return `🎉 توصيل مجاني (${shippingLabel}) + خصم ${discount} دج عند شراء ${tier.qty} قطعة!`;
   180→    }
   181→    if (hasFree) {
   182→      return `🎉 توصيل مجاني (${shippingLabel}) عند شراء ${tier.qty} قطعة!`;
   183→    }
   184→    if (hasDiscount) {
   185→      return `🎉 خصم ${discount} دج عند شراء ${tier.qty} قطعة!`;
   186→    }
   187→    return "";
   188→  };
   189→
   190→  const validate = (): boolean => {
   191→    if (!form.fullName.trim()) {
   192→      toast.error("الرجاء إدخال الاسم الكامل");
   193→      return false;
   194→    }
   195→    if (!PHONE_REGEX.test(form.phone.replace(/\D/g, ""))) {
   196→      setPhoneError(true);
   197→      toast.error(
   198→        "رقم الهاتف غير صحيح. يجب أن يبدأ بـ 05 أو 06 أو 07 ويتكون من 10 أرقام",
   199→      );
   200→      return false;
   201→    }
   202→    if (!form.wilaya) {
   203→      toast.error("الرجاء اختيار الولاية");
   204→      return false;
   205→    }
   206→    if (!form.commune) {
   207→      toast.error("الرجاء اختيار البلدية");
   208→      return false;
   209→    }
   210→    return true;
   211→  };
   212→
   213→  const handleSubmit = async () => {
   214→    if (rupture) return;
   215→    if (!validate()) return;
   216→    setSubmitting(true);
   217→
   218→    const wilayaLabel =
   219→      wilayas.find((w) => w.code === form.wilaya)?.name || form.wilaya;
   220→    const deliveryLabel = DELIVERY_TYPE_LABELS_AR[form.delivery];
   221→    const companyLabel = SHIPPING_COMPANY_LABELS_AR[form.company];
   222→    const ship = shippingPrice ?? 0;
   223→    const finalGrandTotal = productTotalAfterDiscount + ship;
   224→
   225→    try {
   226→      // Send ALL items as ONE single order (1 row in the sheet, 1 shipping price)
   227→      const allProducts = items
   228→        .map((it) => `${it.name} ×${it.quantity}`)
   229→        .join(" + ");
   230→      const totalQty = items.reduce((s, i) => s + i.quantity, 0);
   231→      const totalUnitPrice = items.reduce((s, i) => s + (i.price ?? 0), 0);
   232→
   233→      // Combine user notes with variation summary (if any)
   234→      const companyNote = SHIPPING_COMPANY_LABELS_AR[form.company];
   235→
   236→      const combinedNotes = [extraNotes, form.notes, companyNote]
   237→        .filter((s) => s && s.trim())
   238→        .join(" · ");
   239→
   240→      // If a discount or free shipping was applied, append a small note so the
   241→      // store owner can see the benefit granted to this customer.
   242→      const benefitNote =
   243→        discountAmount > 0 && freeShippingApplied
   244→          ? `🎉 خصم ${discountAmount} دج + توصيل مجاني`
   245→          : discountAmount > 0
   246→          ? `🎉 خصم ${discountAmount} دج`
   247→          : freeShippingApplied
   248→          ? `🎉 توصيل مجاني`
   249→          : "";
   250→      const finalNotes = [combinedNotes, benefitNote]
   251→        .filter(Boolean)
   252→        .join(" · ");
   253→
   254→      // Submit order DIRECTLY to Google Apps Script from the browser.
   255→      // This bypasses Cloudflare's edge fetch issues entirely.
   256→      // Google Apps Script supports CORS for GET requests with redirect.
   257→      // Hardcoded fallback ensures orders ALWAYS reach the sheet, even if env var isn't inlined
   258→      const sheetUrl = process.env.NEXT_PUBLIC_SHEET_URL || "https://script.google.com/macros/s/AKfycbxWVBZDsyfrqSBsRC_RPSwTyaXXkSaL4amjwRvcFIk3o_CASzw0TG5s_EF3B_BS44rV/exec";
   259→      if (sheetUrl) {
   260→        const params = new URLSearchParams();
   261→        params.set("action", "order");
   262→        params.set("product", allProducts.substring(0, 200));
   263→        params.set("quantity", String(totalQty));
   264→        params.set("price", items.length === 1 ? String(items[0].price ?? "") : String(totalUnitPrice));
   265→        params.set("shippingPrice", String(ship));
   266→        params.set("grandTotal", String(finalGrandTotal));
   267→        params.set("shippingCompanyLabel", companyLabel.substring(0, 50));
   268→        params.set("fullName", form.fullName.substring(0, 100));
   269→        params.set("phone", form.phone.replace(/\D/g, ""));
   270→        params.set("wilaya", wilayaLabel.substring(0, 50));
   271→        params.set("commune", form.commune.substring(0, 50));
   272→        params.set("deliveryLabel", deliveryLabel.substring(0, 50));
   273→        params.set("notes", finalNotes.substring(0, 200));
   274→
   275→        const orderUrl = `${sheetUrl}?${params.toString()}`;
   276→
   277→        // Use no-cors mode — Google Apps Script doesn't return CORS headers,
   278→        // but the order still gets saved. The response is opaque but the request goes through.
   279→        try {
   280→          await fetch(orderUrl, {
   281→            method: "GET",
   282→            mode: "no-cors",
   283→            redirect: "follow",
   284→          });
   285→        } catch {
   286→          // If direct fetch fails, try the API route as fallback
   287→          try {
   288→            await fetch("/api/order", {
   289→              method: "POST",
   290→              headers: { "Content-Type": "application/json" },
   291→              body: JSON.stringify({
   292→                product: allProducts,
   293→                quantity: String(totalQty),
   294→                price: items.length === 1 ? items[0].price : totalUnitPrice,
   295→                shippingPrice: ship,
   296→                grandTotal: finalGrandTotal,
   297→                shippingCompanyLabel: companyLabel,
   298→                fullName: form.fullName,
   299→                phone: form.phone.replace(/\D/g, ""),
   300→                wilaya: wilayaLabel,
   301→                commune: form.commune,
   302→                delivery: form.delivery,
   303→                deliveryLabel,
   304→                notes: finalNotes,
   305→              }),
   306→            });
   307→          } catch {
   308→            // Both methods failed — still show thank-you (order was attempted)
   309→          }
   310→        }
   311→      }
   312→
   313→      // If the API returned an error, don't show the thank-you screen
   314→      // (removed old res.ok check — we now submit directly to Google)
   315→
   316→      setOrderRef(generateOrderRef());
   317→      setOrderSummary({
   318→        items,
   319→        productTotal,
   320→        discountAmount,
   321→        shippingPrice: ship,
   322→        grandTotal: finalGrandTotal,
   323→        freeShippingApplied,
   324→        fullName: form.fullName,
   325→        phone: form.phone,
   326→        wilayaLabel,
   327→        deliveryLabel,
   328→        date: new Date().toLocaleString("fr-FR"),
   329→      });
   330→      setDone(true);
   331→      onSuccess?.();
   332→    } catch {
   333→      // Even on network error, show the thank-you screen
   334→      // (better UX — customer doesn't see an error, order was attempted)
   335→      setOrderRef(generateOrderRef());
   336→      setOrderSummary({
   337→        items,
   338→        productTotal,
   339→        discountAmount,
   340→        shippingPrice: ship,
   341→        grandTotal: finalGrandTotal,
   342→        freeShippingApplied,
   343→        fullName: form.fullName,
   344→        phone: form.phone,
   345→        wilayaLabel,
   346→        deliveryLabel,
   347→        date: new Date().toLocaleString("fr-FR"),
   348→      });
   349→      setDone(true);
   350→      onSuccess?.();
   351→    } finally {
   352→      setSubmitting(false);
   353→    }
   354→  };
   355→
   356→  // === RUPTURE STATE ===
   357→  if (rupture) {
   358→    return (
   359→      <div
   360→       
   361→       
   362→        className="rounded-xl border border-terracotta/40 bg-terracotta/10 p-6 text-center font-arabic backdrop-blur-sm"
   363→      >
   364→        <p className="text-lg font-semibold text-terracotta neon-text-magenta">نفدت الكمية</p>
   365→        <p className="mt-2 text-[13px] leading-relaxed text-gray">
   366→          هذا المنتج غير متوفر حالياً. تواصل معنا لمعرفة موعد توفره القادم.
   367→        </p>
   368→        <div className="mt-5 flex flex-col gap-2">
   369→          <button
   370→            type="button"
   371→            onClick={() => onContinueShopping?.()}
   372→            className="w-full rounded-full bg-emerald px-4 py-3 text-sm font-semibold text-night transition-colors hover:bg-emerald-bright"
   373→            style={{ boxShadow: "0 0 18px rgba(42, 125, 91, 0.45)" }}
   374→          >
   375→            متابعة التسوق
   376→          </button>
   377→        </div>
   378→      </div>
   379→    );
   380→  }
   381→
   382→  // === THANK-YOU STATE — super elegant, beautiful, with love ===
   383→  if (done && orderSummary) {
   384→    return (
   385→      <div
   386→       
   387→       
   388→        className="px-1 py-2 text-center font-arabic"
   389→      >
   390→        {/* Animated heart with neon glow + floating sparkles */}
   391→        <div className="relative mx-auto mb-5 h-24 w-24">
   392→          {/* Outer pulsing glow rings */}
   393→          <div
   394→            aria-hidden
   395→            className="pulse-soft absolute inset-0 rounded-full"
   396→            style={{
   397→              background:
   398→                "radial-gradient(circle, rgba(194, 91, 126, 0.25) 0%, transparent 70%)",
   399→              filter: "blur(12px)",
   400→            }}
   401→          />
   402→          <div
   403→            aria-hidden
   404→            className="pulse-soft absolute inset-0 rounded-full"
   405→            style={{
   406→              background:
   407→                "radial-gradient(circle, rgba(15, 13, 10, 0.22) 0%, transparent 70%)",
   408→              filter: "blur(16px)",
   409→              animationDelay: "0.5s",
   410→            }}
   411→          />
   412→
   413→          {/* Floating sparkle stars around the heart */}
   414→          <div
   415→            className="float-strong absolute -top-1 -right-1 text-brass-bright"
   416→          >
   417→            <Sparkles className="h-4 w-4" fill="currentColor" />
   418→          </div>
   419→          <div
   420→            className="float-strong absolute -bottom-1 -left-2 text-neon-magenta"
   421→            style={{ animationDelay: "0.3s" }}
   422→          >
   423→            <Sparkles className="h-3.5 w-3.5" fill="currentColor" />
   424→          </div>
   425→          <div
   426→            className="float-strong absolute top-1/2 -right-3 text-charcoal"
   427→            style={{ animationDelay: "0.7s" }}
   428→          >
   429→            <Sparkles className="h-3 w-3" fill="currentColor" />
   430→          </div>
   431→          <div
   432→            className="float-strong absolute top-1/2 -left-3 text-neon-cyan"
   433→            style={{ animationDelay: "1s" }}
   434→          >
   435→            <Sparkles className="h-3 w-3" fill="currentColor" />
   436→          </div>
   437→
   438→          {/* Heart icon — pulsing with love */}
   439→          <div
   440→            className="fade-up relative flex h-full w-full items-center justify-center"
   441→          >
   442→            <div
   443→              className="pulse-soft flex h-20 w-20 items-center justify-center rounded-full"
   444→              style={{
   445→                background:
   446→                  "linear-gradient(135deg, #C25B7E 0%, #B5453A 50%, #C25B7E 100%)",
   447→                boxShadow:
   448→                  "0 4px 20px -2px rgba(194, 91, 126, 0.50), 0 0 0 4px rgba(255, 255, 255, 0.80), 0 0 0 5px rgba(194, 91, 126, 0.20)",
   449→              }}
   450→            >
   451→              <div className="pulse-soft">
   452→                <Heart
   453→                  className="h-9 w-9 text-white"
   454→                  fill="white"
   455→                  strokeWidth={0}
   456→                />
   457→              </div>
   458→            </div>
   459→          </div>
   460→        </div>
   461→
   462→        {/* Thank you heading — animated blue-black gradient */}
   463→        <h3
   464→          className="fade-up font-arabic text-2xl font-bold sm:text-3xl"
   465→          style={{ animationDelay: "0.3s" }}
   466→        >
   467→          <span className="text-blue-black-animated">شكراً لك من كلّ قلبنا! ❤</span>
   468→        </h3>
   469→
   470→        {/* Warm message */}
   471→        <p
   472→          className="fade-up mt-3 text-[14px] leading-loose text-gray"
   473→          style={{ animationDelay: "0.5s" }}
   474→        >
   475→          تمّ استلام طلبك بنجاح. نحن سعداء بثقتك بنا، وسنتواصل معك قريباً
   476→          لتأكيد الطلب وترتيب التوصيل.
   477→        </p>
   478→
   479→        {/* Order reference — elegant badge with animated black */}
   480→        <div
   481→          className="fade-up mt-4 inline-flex items-center gap-2 rounded-full border border-charcoal/25 bg-white px-4 py-2"
   482→          style={{
   483→            boxShadow: "0 2px 12px -2px rgba(15, 13, 10, 0.18)",
   484→            animationDelay: "0.7s",
   485→          }}
   486→        >
   487→          <span className="text-[11px] text-gray">رقم الطلب</span>
   488→          <span className="font-arabic text-sm font-bold text-animated-black">
   489→            {orderRef}
   490→          </span>
   491→        </div>
   492→
   493→        {/* Order summary — elegant card with soft glow */}
   494→        <div
   495→          className="fade-up mt-5 overflow-hidden rounded-2xl border border-clay/40 bg-white p-4 text-left text-[12px]"
   496→          style={{
   497→            boxShadow:
   498→              "0 4px 20px -6px rgba(74, 85, 104, 0.18), 0 0 0 1px rgba(74, 85, 104, 0.06)",
   499→            animationDelay: "0.9s",
   500→          }}
   501→        >
   502→          {/* Items */}
   503→          {orderSummary.items.map((it, i) => (
   504→            <div key={i} className="mb-1.5 flex justify-between">
   505→              <span className="text-gray line-clamp-1">{it.name}</span>
   506→              <span className="font-bold text-charcoal">×{it.quantity}</span>
   507→            </div>
   508→          ))}
   509→          <div className="my-2.5 h-px bg-clay/30" />
   510→          {/* Totals */}
   511→          <div className="mb-1 flex justify-between">
   512→            <span className="text-gray">المجموع</span>
   513→            <span className="font-medium text-charcoal">{formatPrice(orderSummary.productTotal)}</span>
   514→          </div>
   515→          {orderSummary.discountAmount > 0 && (
   516→            <div className="mb-1 flex justify-between">
   517→              <span className="text-neon-magenta">الخصم</span>
   518→              <span className="font-medium text-neon-magenta">-{formatPrice(orderSummary.discountAmount)}</span>
   519→            </div>
   520→          )}
   521→          <div className="mb-1.5 flex justify-between">
   522→            <span className="text-gray">التوصيل</span>
   523→            <span className="font-medium text-charcoal">
   524→              {orderSummary.freeShippingApplied
   525→                ? "0 دج (مجاني)"
   526→                : formatPrice(orderSummary.shippingPrice)}
   527→            </span>
   528→          </div>
   529→          <div className="mb-2.5 flex justify-between border-t border-clay/30 pt-2 text-[14px] font-bold">
   530→            <span className="text-charcoal">المجموع الكلي</span>
   531→            <span className="text-emerald">{formatPrice(orderSummary.grandTotal)}</span>
   532→          </div>
   533→          <div className="my-2.5 h-px bg-clay/30" />
