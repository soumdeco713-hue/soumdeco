    57→export function CodOrderForm({
    58→  items: initialItems,
    59→  rupture,
    60→  onSuccess,
    61→  onContinueShopping,
    62→  extraNotes,
    63→  quantityTiers,
    64→  variantsMissing = false,
    65→  missingVariantNames = [],
    66→}: CodOrderFormProps) {
    67→  const { wilayas, getCommunesForWilaya, loading } = useAlgeriaData();
    68→  const [items, setItems] = useState<OrderItem[]>(initialItems);
    69→  const [form, setForm] = useState<FormState>({
    70→    fullName: "",
    71→    phone: "",
    72→    wilaya: "",
    73→    commune: "",
    74→    company: "zr_express",
    75→    delivery: "stop_desk",
    76→    notes: "",
    77→  });
    78→  const [phoneError, setPhoneError] = useState(false);
    79→  const [submitting, setSubmitting] = useState(false);
    80→  const [done, setDone] = useState(false);
    81→  const [orderRef, setOrderRef] = useState("");
    82→  const [customQty, setCustomQty] = useState(""); // local input text — independent from buttons
    83→  const [orderSummary, setOrderSummary] = useState<null | {
    84→    items: OrderItem[];
    85→    productTotal: number;
    86→    discountAmount: number;
    87→    shippingPrice: number;
    88→    grandTotal: number;
    89→    freeShippingApplied: boolean;
    90→    fullName: string;
    91→    phone: string;
    92→    wilayaLabel: string;
    93→    deliveryLabel: string;
    94→    date: string;
    95→  }>(null);
    96→
    97→  // Sync items when prop changes (e.g., modal opens different product)
    98→  useEffect(() => {
    99→    setItems(initialItems);
   100→  }, [initialItems]);
   101→
   102→  const communes = useMemo(
   103→    () => (form.wilaya ? getCommunesForWilaya(form.wilaya) : []),
   104→    [form.wilaya, getCommunesForWilaya],
   105→  );
   106→
   107→  // Reset commune when wilaya changes
   108→  useEffect(() => {
   109→    setForm((f) => ({ ...f, commune: "" }));
   110→  }, [form.wilaya]);
   111→
   112→  // The NORMAL shipping price (without any tier benefit applied).
   113→  // Returns null until BOTH wilaya AND commune are selected.
   114→  const normalShippingPrice = useMemo(() => {
   115→    if (!form.wilaya || !form.commune) return null;
   116→    const result = getShippingPrice(form.wilaya, form.company, form.delivery);
   117→    return result ? result.price : null;
   118→  }, [form.wilaya, form.commune, form.company, form.delivery]);
   119→
   120→  // Active tier — only for single-item orders where the parent passed `quantityTiers`.
   121→  // Used to compute discount and free-shipping benefits.
   122→  //
   123→  // Tier matching logic:
   124→  //  - mode "exact": triggers ONLY when qty === t.qty
   125→  //  - mode "min":   triggers when qty >= t.qty
   126→  //
   127→  // When multiple tiers match (e.g. qty=5 matches both "min:2" and "min:4"),
   128→  // we pick the one with the HIGHEST qualifying qty (most generous threshold reached).
   129→  // Among same-qty tiers, "min" wins over "exact" (more generous).
   130→  const activeTier = useMemo(() => {
   131→    if (items.length !== 1) return null;
   132→    const q = items[0].quantity;
   133→    const allTiers = quantityTiers ?? [];
   134→    if (allTiers.length === 0) return null;
   135→
   136→    // Find all matching tiers
   137→    const matching = allTiers.filter((t) => {
   138→      const mode = t.mode ?? "exact";
   139→      return mode === "min" ? q >= t.qty : q === t.qty;
   140→    });
   141→
   142→    if (matching.length === 0) return null;
   143→    if (matching.length === 1) return matching[0];
   144→
   145→    // Multiple matches — pick the best one.
   146→    // Priority: highest qty threshold first (most generous threshold reached).
   147→    // Among same-qty tiers: "min" beats "exact", then highest discount.
   148→    matching.sort((a, b) => {
   149→      if (a.qty !== b.qty) return b.qty - a.qty; // higher qty first
   150→      const aMode = a.mode ?? "exact";
   151→      const bMode = b.mode ?? "exact";
   152→      if (aMode !== bMode) return aMode === "min" ? -1 : 1; // min first
   153→      const aDisc = a.discountAmount ?? 0;
   154→      const bDisc = b.discountAmount ?? 0;
   155→      return bDisc - aDisc; // higher discount first
   156→    });
   157→    return matching[0];
   158→  }, [items, quantityTiers]);
   159→
   160→  // Discount from the active tier (if any) — clamped so total can never go negative.
   161→  const discountAmount = activeTier?.discountAmount ?? 0;
   162→
   163→  // Whether the active tier grants free shipping for the current delivery type.
   164→  const freeShippingApplied = useMemo(() => {
   165→    if (!activeTier || activeTier.freeShipping === "none") return false;
   166→    if (activeTier.freeShipping === "both") return true;
   167→    if (activeTier.freeShipping === "desk" && form.delivery === "stop_desk") return true;
   168→    if (activeTier.freeShipping === "home" && form.delivery === "home") return true;
   169→    return false;
   170→  }, [activeTier, form.delivery]);
   171→
   172→  const productTotal = useMemo(
   173→    () => items.reduce((s, i) => s + (i.price ?? 0) * i.quantity, 0),
   174→    [items],
   175→  );
   176→
   177→  // Product total AFTER discount (never negative).
   178→  const productTotalAfterDiscount = useMemo(
   179→    () => Math.max(0, productTotal - discountAmount),
   180→    [productTotal, discountAmount],
   181→  );
   182→
   183→  // Final shipping price: 0 if the tier grants free shipping, otherwise normal price.
   184→  const shippingPrice = useMemo(() => {
   185→    if (normalShippingPrice === null) return null;
   186→    return freeShippingApplied ? 0 : normalShippingPrice;
   187→  }, [normalShippingPrice, freeShippingApplied]);
   188→
   189→  const grandTotal = useMemo(() => {
   190→    const ship = shippingPrice ?? 0;
   191→    return productTotalAfterDiscount + ship;
   192→  }, [productTotalAfterDiscount, shippingPrice]);
   193→
   194→  const set = <K extends keyof FormState>(key: K, value: FormState[K]) => {
   195→    setForm((f) => ({ ...f, [key]: value }));
   196→    if (key === "phone") setPhoneError(false);
   197→  };
   198→
   199→  const setSingleQty = (q: number) => {
   200→    if (q < 1) return;
   201→    setItems((prev) =>
   202→      prev.length === 1 ? [{ ...prev[0], quantity: q }] : prev,
   203→    );
   204→  };
   205→
   206→  const tierBenefitLine = (tier: QuantityTier): string => {
   207→    const hasFree = tier.freeShipping && tier.freeShipping !== "none";
   208→    const discount = tier.discountAmount ?? 0;
   209→    const hasDiscount = discount > 0;
   210→    const mode = tier.mode ?? "exact";
   211→    const qtyLabel = mode === "min" ? `${tier.qty} قطعة أو أكثر` : `${tier.qty} قطعة`;
   212→    const shippingLabel =
   213→      tier.freeShipping === "both"
   214→        ? "كلاهما"
   215→        : tier.freeShipping === "desk"
   216→        ? "المكتب"
   217→        : tier.freeShipping === "home"
   218→        ? "المنزل"
   219→        : "";
   220→    if (hasFree && hasDiscount) {
   221→      return `🎉 توصيل مجاني (${shippingLabel}) + خصم ${discount} دج عند شراء ${qtyLabel}!`;
   222→    }
   223→    if (hasFree) {
   224→      return `🎉 توصيل مجاني (${shippingLabel}) عند شراء ${qtyLabel}!`;
   225→    }
   226→    if (hasDiscount) {
   227→      return `🎉 خصم ${discount} دج عند شراء ${qtyLabel}!`;
   228→    }
   229→    return "";
   230→  };
   231→
   232→  const validate = (): boolean => {
   233→    if (!form.fullName.trim()) {
   234→      toast.error("الرجاء إدخال الاسم الكامل");
   235→      return false;
   236→    }
   237→    if (!PHONE_REGEX.test(form.phone.replace(/\D/g, ""))) {
   238→      setPhoneError(true);
   239→      toast.error(
   240→        "رقم الهاتف غير صحيح. يجب أن يبدأ بـ 05 أو 06 أو 07 ويتكون من 10 أرقام",
   241→      );
   242→      return false;
   243→    }
   244→    if (!form.wilaya) {
   245→      toast.error("الرجاء اختيار الولاية");
   246→      return false;
   247→    }
   248→    if (!form.commune) {
   249→      toast.error("الرجاء اختيار البلدية");
   250→      return false;
   251→    }
   252→    return true;
   253→  };
   254→
   255→  const handleSubmit = async () => {
   256→    if (rupture) return;
   257→
   258→    // OBLIGATORY VARIANT CHECK — same priority as name field.
   259→    // If the product has variants and the customer hasn't selected them,
   260→    // block the order submission. This catches the case where the customer
   261→    // fills out the order form directly on the product page without
   262→    // selecting variants first.
   263→    if (variantsMissing) {
   264→      if (missingVariantNames.length === 1) {
   265→        toast.error(`الرجاء اختيار ${missingVariantNames[0]}`);
   266→      } else {
   267→        toast.error(`الرجاء اختيار: ${missingVariantNames.join("، ")}`);
   268→      }
   269→      return;
   270→    }
   271→
   272→    if (!validate()) return;
   273→
   274→    // P0 FIX #2: Re-check rupture at checkout submit time
   275→    if (typeof rupture !== "undefined" && rupture) {
   276→      toast.error("هذا المنتج نفدت الكمية. لا يمكن إتمام الطلب.");
   277→      return;
   278→    }
   279→
   280→    // P0 FIX #4: Sanitize items — remove NaN prices, prevent NaN from reaching Apps Script
   281→    const sanitizedItems = items.map((it) => ({
   282→      ...it,
   283→      price:
   284→        typeof it.price === "number" && !isNaN(it.price) && it.price >= 0
   285→          ? it.price
   286→          : null, // null = price-on-request (prevents NaN in sheet)
   287→      name: it.name || "منتج بدون اسم", // prevent empty names
   288→    }));
   289→
   290→    // P0 FIX #3: Filter out items with no name (orphan/invalid cart items)
   291→    // Uses productId if available, otherwise falls back to name check
   292→    const validItems = sanitizedItems.filter(
   293→      (it) => it.name && it.name.trim() !== "",
   294→    );
   295→    if (validItems.length === 0) {
   296→      toast.error("السلة فارغة أو تحتوي على منتجات لم تعد متاحة.");
   297→      return;
   298→    }
   299→
   300→    // SECONDARY VARIANT CHECK: ensure every item that requires a variant
   301→    // actually has variant info in its name (format: "Product Name (variant info)")
   302→    // This is a safety net — the primary check is on the product page before
   303→    // add-to-cart. But if somehow an item gets into the cart without variant
   304→    // info (e.g., old cart from before this feature), we catch it here.
   305→    const itemsMissingVariant = validItems.filter((it) => {
   306→      // If the product name contains "(" it means variant info was included
   307→      // If it does NOT contain "(", the product either has no variants OR
   308→      // the variant was not selected (which shouldn't happen due to the
   309→      // primary check, but this is a safety net)
   310→      // We can't know here if the product has variants (we don't have the
   311→      // catalog), so we just check: if productId exists and name has no "(",
   312→      // it MIGHT be missing variant info. But since we can't verify without
   313→      // the catalog, we skip this check for multi-item carts and only
   314→      // apply it for single-item orders where we have the variantTiers prop.
   315→      return false; // safety net disabled — primary check on product page is sufficient
   316→    });
   317→    if (itemsMissingVariant.length > 0) {
   318→      toast.error("الرجاء اختيار الخيارات المطلوبة للمنتج.");
   319→      return;
   320→    }
   321→
   322→    setSubmitting(true);
   323→
   324→    const wilayaLabel =
   325→      wilayas.find((w) => w.code === form.wilaya)?.name || form.wilaya;
   326→    const deliveryLabel = DELIVERY_TYPE_LABELS_AR[form.delivery];
   327→    const companyLabel = SHIPPING_COMPANY_LABELS_AR[form.company];
   328→    const ship = shippingPrice ?? 0;
   329→    const finalGrandTotal = productTotalAfterDiscount + ship;
   330→
   331→    try {
   332→      // Send ALL items as ONE single order (1 row in the sheet, 1 shipping price)
   333→      const allProducts = validItems
   334→        .map((it) => `${it.name} ×${it.quantity}`)
   335→        .join(" + ");
   336→      const totalQty = validItems.reduce((s, i) => s + i.quantity, 0);
   337→      const totalUnitPrice = validItems.reduce(
   338→        (s, i) => s + (i.price ?? 0),
   339→        0,
   340→      );
   341→
   342→      // ============================================================
   343→      //  VARIANT EXTRACTION — extract clean variant name for the
   344→      //  new "Variant" column in the Orders sheet.
   345→      //
   346→      //  Cart item names look like:
   347→      //    "Cocotte minute (اللون: Red)" → variant = "Red"
   348→      //    "Product (اللون: Red · المقاس: Large)" → variant = "Red - Large"
   349→      //    "Simple Product" → variant = "" (no variant)
   350→      //
   351→      //  This is sent as a separate URL param (&variant=Red) so the
   352→      //  Apps Script writes it to the Variant column — zero parsing
   353→      //  server-side. The trigger reads the column directly.
   354→      // ============================================================
   355→      const extractVariant = (itemName: string): string => {
   356→        const match = itemName.match(/\(([^)]+)\)\s*$/);
   357→        if (!match) return "";
   358→        const content = match[1].trim();
   359→        const parts = content.split("·");
   360→        const values: string[] = [];
   361→        for (const part of parts) {
   362→          const trimmed = part.trim();
   363→          const colonIdx = trimmed.lastIndexOf(":");
   364→          if (colonIdx >= 0) {
   365→            const value = trimmed.substring(colonIdx + 1).trim();
   366→            if (value) values.push(value);
   367→          } else if (trimmed) {
   368→            values.push(trimmed);
   369→          }
   370→        }
   371→        return values.join(" - ");
   372→      };
   373→      // For single-item orders: extract variant directly
   374→      // For multi-item orders: join all variants (rare but handles it)
   375→      const orderVariant = validItems.length === 1
   376→        ? extractVariant(validItems[0].name)
   377→        : validItems
   378→            .map((it) => {
   379→              const v = extractVariant(it.name);
   380→              return v ? `${it.name.split(" (")[0]}: ${v}` : "";
   381→            })
   382→            .filter(Boolean)
   383→            .join(" + ");
   384→
   385→      // Combine user notes with variation summary (if any)
   386→      const companyNote = SHIPPING_COMPANY_LABELS_AR[form.company];
   387→
   388→      const combinedNotes = [extraNotes, form.notes, companyNote]
   389→        .filter((s) => s && s.trim())
   390→        .join(" · ");
   391→
   392→      // If a discount or free shipping was applied, append a small note so the
   393→      // store owner can see the benefit granted to this customer.
   394→      const benefitNote =
   395→        discountAmount > 0 && freeShippingApplied
   396→          ? `🎉 خصم ${discountAmount} دج + توصيل مجاني`
   397→          : discountAmount > 0
   398→          ? `🎉 خصم ${discountAmount} دج`
   399→          : freeShippingApplied
   400→          ? `🎉 توصيل مجاني`
   401→          : "";
   402→      const finalNotes = [combinedNotes, benefitNote]
   403→        .filter(Boolean)
   404→        .join(" · ");
   405→
   406→      // Submit order DIRECTLY to Google Apps Script using the bulletproof
   407→      // clientSubmitOrder function (real CORS mode, retry logic, timeout).
   408→      // This bypasses Cloudflare's broken edge API entirely.
   409→      const { clientSubmitOrder } = await import("@/lib/client-sheet");
   410→
   411→      // ============================================================
   412→      //  BUILD STOCK KEY — the EXACT Stock tab entry name
   413→      //  for variants that have stock tracking (|stock in variants string).
   414→      //
   415→      //  Format: "ProductName - VariantName"
   416→      //  This matches EXACTLY what updateVariantStockTab_ creates in the
   417→      //  Stock tab. The SUMIFS formula in the Stock tab uses this key
   418→      //  to auto-decrement when order status = "Confirmed".
   419→      //
   420→      //  Only built for variants that HAVE |stock set.
   421→      //  If no variant has stock → stockKey is empty → no variant decrement.
   422→      // ============================================================
   423→      let stockKey = "";
   424→      if (validItems.length === 1) {
   425→        const item = validItems[0];
   426→        // Extract product name (strip variant parentheses)
   427→        const prodName = item.name.replace(/\s*\([^)]+\)\s*$/, "").trim();
   428→        // Try to find the product in the catalog to get its variants
   429→        // The item might have variant info in its name
   430→        const variantMatch = item.name.match(/\(([^)]+)\)\s*$/);
   431→        if (variantMatch) {
   432→          const variantContent = variantMatch[1].trim();
   433→          const variantParts = variantContent.split("·");
   434→          // For each selected variant, extract the value
   435→          const selectedValues: string[] = [];
   436→          for (const part of variantParts) {
   437→            const trimmed = part.trim();
   438→            const colonIdx = trimmed.lastIndexOf(":");
   439→            if (colonIdx >= 0) {
   440→              const value = trimmed.substring(colonIdx + 1).trim();
   441→              if (value) selectedValues.push(value);
   442→            } else if (trimmed) {
   443→              selectedValues.push(trimmed);
   444→            }
   445→          }
   446→          // Build stockKey for each variant value
   447→          // The Stock tab has entries like "Product - 10L" (one per variant)
   448→          // We need to try each selected variant to find which one has stock
   449→          // Send ALL possible keys (comma-separated) — the SUMIFS will match
   450→          // the one that exists in the Stock tab
   451→          if (selectedValues.length > 0) {
   452→            stockKey = selectedValues
   453→              .map((v) => `${prodName} - ${v}`)
   454→              .join(",");
   455→          }
   456→        }
   457→      }
   458→
   459→      const orderOk = await clientSubmitOrder({
   460→        product: allProducts,
   461→        quantity: String(totalQty),
   462→        price: items.length === 1 ? items[0].price ?? null : totalUnitPrice,
   463→        shippingPrice: ship,
   464→        grandTotal: finalGrandTotal,
   465→        shippingCompanyLabel: companyLabel,
   466→        fullName: form.fullName,
   467→        phone: form.phone.replace(/\D/g, ""),
   468→        wilaya: wilayaLabel,
   469→        commune: form.commune,
   470→        deliveryLabel,
   471→        notes: finalNotes,
   472→        variant: orderVariant,
   473→        stockKey,
   474→      });
   475→
   476→      if (!orderOk) {
   477→        // Order failed to reach the sheet — save to the retry queue.
   478→        // The retry queue is processed on the next page visit (in useCatalog init).
   479→        // The customer already saw a thank-you screen — this is a background safety net.
   480→        try {
   481→          const { addFailedOrder } = await import("@/lib/failed-orders");
   482→          addFailedOrder({
   483→            timestamp: new Date().toISOString(),
   484→            product: allProducts,
   485→            quantity: String(totalQty),
   486→            price: items.length === 1 ? items[0].price ?? null : totalUnitPrice,
   487→            shippingPrice: ship,
   488→            grandTotal: finalGrandTotal,
   489→            shippingCompanyLabel: companyLabel,
   490→            fullName: form.fullName,
   491→            phone: form.phone,
   492→            wilaya: wilayaLabel,
   493→            commune: form.commune,
   494→            deliveryLabel,
   495→            notes: finalNotes,
   496→          });
   497→        } catch {
   498→          console.error(
   499→            "[Order] CRITICAL: Failed to submit order AND failed to save to retry queue",
   500→          );
   501→        }
   502→      }
   503→
   504→      // ALWAYS show the thank-you screen — the customer has done their part.
   505→      // The order is either in the sheet OR in localStorage for retry.
   506→      setOrderRef(generateOrderRef());
   507→      setOrderSummary({
   508→        items,
   509→        productTotal,
   510→        discountAmount,
   511→        shippingPrice: ship,
   512→        grandTotal: finalGrandTotal,
   513→        freeShippingApplied,
   514→        fullName: form.fullName,
   515→        phone: form.phone,
   516→        wilayaLabel,
   517→        deliveryLabel,
   518→        date: new Date().toLocaleString("fr-FR"),
   519→      });
   520→      setDone(true);
   521→      onSuccess?.();
   522→
   523→      // Send Telegram notification (NON-BLOCKING but awaited for reliability)
   524→      // Fires AFTER the order is confirmed — never blocks the thank-you screen
   525→      // but DOES wait for completion so the fetch isn't cancelled if the
   526→      // customer navigates away quickly.
   527→      try {
   528→        const { sendOrderTelegramNotification } = await import("@/lib/telegram-notify");
   529→        await sendOrderTelegramNotification().catch(() => {});
   530→      } catch {
   531→        // Silent — Telegram is optional
   532→      }
   533→    } catch {
   534→      // Even on unhandled exception, show the thank-you screen
   535→      // (better UX — customer doesn't see an error, order was attempted)
