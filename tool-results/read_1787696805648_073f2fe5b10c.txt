    46→    46→    return jsonOut({ ok: false, error: 'unknown action: ' + action });

    97→    97→function serveStock() {
    98→    98→  var ss = SpreadsheetApp.getActiveSpreadsheet();
    99→    99→  var sheet = ss.getSheetByName(STOCK_SHEET);
   100→   100→  if (!sheet) { sheet = ss.insertSheet(STOCK_SHEET); sheet.appendRow(['Product Name','Stock Count']); }
   101→   101→  var headerRow = sheet.getRange(1, 1, 1, 2).getValues()[0];
   102→   102→  if (String(headerRow[0]||'') !== 'Product Name' || String(headerRow[1]||'') !== 'Stock Count') {
   103→   103→    sheet.getRange(1, 1, 1, 2).setValues([['Product Name','Stock Count']]);
   104→   104→  }
   105→   105→  var values = sheet.getDataRange().getValues();
   106→   106→  var rows = [];
   107→   107→  for (var i = 0; i < values.length; i++) {
   108→   108→    var r = values[i];
   109→   109→    if (i === 1 && r[0] && /[\u0600-\u06FF\u{1F000}-\u{1FFFF}]/u.test(String(r[0]))) continue;
   110→   110→    rows.push(r.map(function(c){ return '"' + String(c == null ? '' : c).replace(/"/g, '""') + '"'; }).join(','));
   111→   111→  }
   112→   112→  var csv = rows.join('\n');
   113→   113→  return ContentService.createTextOutput(csv).setMimeType(ContentService.MimeType.CSV);
   114→   114→}

   140→   140→function serveProducts() {
   141→   141→  var sheet = ensureProductsSheet();
   142→   142→  var values = sheet.getDataRange().getValues();
   143→   143→  if (values.length < 2) return jsonOut([]);
   144→   144→  var header = values[0];
   145→   145→  var out = [];
   146→   146→  // Track seen IDs to skip duplicate rows (the sheet sometimes has dupes
   147→   147→  // when a product is saved twice or moved between categories).
   148→   148→  var seenIds = {};
   149→   149→  // Start from row 2 (index 1). Skip guidance row (row 2) if it contains
   150→   150→  // Arabic/emoji hints instead of real product data.
   151→   151→  for (var i = 1; i < values.length; i++) {
   152→   152→    var r = values[i];
   153→   153→    if (!r[0]) continue;
   154→   154→    // Skip the guidance row — detect it by checking if the id cell contains
   155→   155→    // non-ASCII characters (emojis/Arabic) which real product IDs never have.
   156→   156→    var idStr = String(r[0]);
   157→   157→    if (/[\u0600-\u06FF\u{1F000}-\u{1FFFF}]/u.test(idStr)) continue;
   158→   158→    // Skip duplicate IDs (keep first occurrence)
   159→   159→    if (seenIds[idStr]) continue;
   160→   160→    seenIds[idStr] = true;
   161→   161→    var obj = {};
   162→   162→    for (var j = 0; j < header.length; j++) obj[header[j]] = r[j];
   163→   163→    obj.price = (obj.price===''||obj.price===null||obj.price===undefined)?null:Number(obj.price);
   164→   164→    obj.oldPrice = (obj.oldPrice===''||obj.oldPrice===null||obj.oldPrice===undefined)?null:Number(obj.oldPrice);
   165→   165→    obj.featured = (obj.featured===true||obj.featured===1||obj.featured==='1'||(typeof obj.featured==='string'&&obj.featured.toLowerCase()==='true'));
   166→   166→    obj.isSpecialOffer = (obj.isSpecialOffer===true||obj.isSpecialOffer===1||obj.isSpecialOffer==='1'||(typeof obj.isSpecialOffer==='string'&&obj.isSpecialOffer.toLowerCase()==='true'));
   167→   167→    // stock — null when blank/non-numeric, otherwise a number.
   168→   168→    obj.stock = (obj.stock===''||obj.stock===null||obj.stock===undefined)?null:Number(obj.stock);
   169→   169→    // variants — pass through as a string for the client to parse.
   170→   170→    obj.variants = obj.variants==null?'':String(obj.variants);
   171→   171→    if ((!obj.images||String(obj.images).trim()==='')&&obj.image) obj.images = String(obj.image);
   172→   172→    // Auto-fix common category typos
   173→   173→    if (obj.category) {
   174→   174→      var cat = String(obj.category).trim();
   175→   175→      if (cat === 'Meubes') cat = 'Meubles';
   176→   176→      obj.category = cat;
   177→   177→    }
   178→   178→    out.push(obj);
   179→   179→  }
   180→   180→  return jsonOut(out);
   181→   181→}

   183→   183→function doCreateProduct(p) {
   184→   184→  var sheet = ensureProductsSheet();
   185→   185→  if (findProductRow_(sheet, p.id) >= 0) return doUpdateProduct(p);
   186→   186→  sheet.appendRow(buildProductRow_(p));
   187→   187→  // Auto-add product name to Stock tab so admin only needs to set the stock count
   188→   188→  addToStockTab_(p.name || '');
   189→   189→  // Also sync stock from product to Stock tab (if admin set a stock value)
   190→   190→  updateStockTab_(p.name || '', p.stock);
   191→   191→  // Also sync per-variant stock to Stock tab (if variants have stock values)
   192→   192→  updateVariantStockTab_(p.name || '', p.variants || '');

   256→   256→// ============================================================
   257→   257→//  updateStockTab_ — Sync stock from product to Stock tab
   258→   258→// ============================================================
   259→   259→//  Called from doCreateProduct + doUpdateProduct.
   260→   260→//  If the product has a stock value (number, including 0),
   261→   261→//  this updates (or creates) the corresponding row in the
   262→   262→//  Stock tab so the visitor sees the correct stock count.
   263→   263→//
   264→   264→//  - If productName already exists in Stock tab → update count
   265→   265→//  - If productName doesn't exist → add new row with count
   266→   266→//  - If stock is null/undefined/empty → skip (don't touch Stock tab)
   267→   267→//  - Never touches other products' rows (safe — no overlap)
   268→   268→// ============================================================
   269→   269→function updateStockTab_(productName, stockValue) {
   270→   270→  if (!productName) return;
   271→   271→  // Only update if stock is explicitly set (number, including 0)
   272→   272→  // null/undefined/empty → admin didn't set stock → don't touch Stock tab
   273→   273→  if (stockValue === null || stockValue === undefined || stockValue === '') return;
   274→   274→
   275→   275→  var ss = SpreadsheetApp.getActiveSpreadsheet();
   276→   276→  var sheet = ss.getSheetByName(STOCK_SHEET);
   277→   277→  if (!sheet) {
   278→   278→    // Stock sheet doesn't exist — create it with headers
   279→   279→    sheet = ss.insertSheet(STOCK_SHEET);
   280→   280→    sheet.appendRow(['Product Name', 'Stock Count']);
   281→   281→  }
   282→   282→
   283→   283→  var count = String(stockValue); // number → string for sheet storage
   284→   284→
   285→   285→  // Search for existing row with this product name
   286→   286→  var lastRow = sheet.getLastRow();
   287→   287→  if (lastRow >= 2) {
   288→   288→    var names = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
   289→   289→    for (var i = 0; i < names.length; i++) {
   290→   290→      if (String(names[i][0] || '').trim() === productName.trim()) {
   291→   291→        // Found — update the count (column B = column 2)
   292→   292→        sheet.getRange(i + 2, 2).setValue(count);
   293→   293→        return; // updated, done
   294→   294→      }
   295→   295→    }
   296→   296→  }
   297→   297→  // Not found — add new row
   298→   298→  sheet.appendRow([productName, count]);
   299→   299→}

   301→   301→// ============================================================
   302→   302→//  updateVariantStockTab_ — Sync per-variant stock to Stock tab
   303→   303→// ============================================================
   304→   304→//  Called from doCreateProduct + doUpdateProduct.
   305→   305→//  Parses the variants string and for each variant that has
   306→   306→//  a stock value, creates/updates a row in the Stock tab
   307→   307→//  with name "ProductName - VariantName" and the stock count.
   308→   308→//
   309→   309→//  Variants format: "color:Red:0|0,color:Blue:0|3,color:Green:0"
   310→   310→//  The |stock suffix is optional — only variants WITH it get
   311→   311→//  a Stock tab entry.
   312→   312→//
   313→   313→//  - Format: "ProductName - VariantName","stockCount"
   314→   314→//  - Safe: only touches rows for THIS product's variants
   315→   315→//  - Never overwrites other products' rows
   316→   316→// ============================================================
   317→   317→function updateVariantStockTab_(productName, variantsStr) {
   318→   318→  if (!productName || !variantsStr) return;
   319→   319→
   320→   320→  var ss = SpreadsheetApp.getActiveSpreadsheet();
   321→   321→  var sheet = ss.getSheetByName(STOCK_SHEET);
   322→   322→  if (!sheet) return;
   323→   323→
   324→   324→  // Parse variants string: "color:Red:0|0,color:Blue:0|3,..."
   325→   325→  var parts = variantsStr.split(',');
   326→   326→  for (var i = 0; i < parts.length; i++) {
   327→   327→    var part = parts[i].trim();
   328→   328→    if (!part) continue;
   329→   329→
   330→   330→    // Check for |stock suffix
   331→   331→    var pipeIdx = part.lastIndexOf('|');
   332→   332→    if (pipeIdx < 0) continue; // no stock set for this variant
   333→   333→
   334→   334→    var stockStr = part.substring(pipeIdx + 1).trim();
   335→   335→    var mainPart = part.substring(0, pipeIdx).trim();
   336→   336→
   337→   337→    // Parse stock value (must be a number >= 0)
   338→   338→    var stockNum = Number(stockStr);
   339→   339→    if (isNaN(stockNum) || stockNum < 0) continue;
   340→   340→
   341→   341→    // Extract variant name from mainPart: "type:name:priceAdjustment"
   342→   342→    // The name is between the first and last colon
   343→   343→    var firstColon = mainPart.indexOf(':');
   344→   344→    if (firstColon < 0) continue;
   345→   345→    var rest = mainPart.substring(firstColon + 1);
   346→   346→    var lastColon = rest.lastIndexOf(':');
   347→   347→    var variantName;
   348→   348→    if (lastColon < 0) {
   349→   349→      variantName = rest.trim();
   350→   350→    } else {
   351→   351→      variantName = rest.substring(0, lastColon).trim();
   352→   352→    }
   353→   353→    if (!variantName) continue;
   354→   354→
   355→   355→    // Build the Stock tab entry name: "ProductName - VariantName"
   356→   356→    var stockTabName = productName + ' - ' + variantName;
   357→   357→
   358→   358→    // Search for existing row and update, or add new
   359→   359→    var lastRow = sheet.getLastRow();
   360→   360→    var found = false;
   361→   361→    if (lastRow >= 2) {
   362→   362→      var names = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
   363→   363→      for (var j = 0; j < names.length; j++) {
   364→   364→        if (String(names[j][0] || '').trim() === stockTabName) {
   365→   365→          // Found — update the count
   366→   366→          sheet.getRange(j + 2, 2).setValue(String(stockNum));
   367→   367→          found = true;
   368→   368→          break;
   369→   369→        }
   370→   370→      }
   371→   371→    }
   372→   372→    if (!found) {
   373→   373→      // Not found — add new row
   374→   374→      sheet.appendRow([stockTabName, String(stockNum)]);
   375→   375→    }
   376→   376→  }
   377→   377→}

   388→   388→function buildProductRow_(p) {
   389→   389→  var imagesStr = String(p.images || p.image || '');
   390→   390→  var coverImage = imagesStr ? imagesStr.split(IMG_SEP)[0] : '';
   391→   391→  return PRODUCTS_COLS.map(function(col) {
   392→   392→    if (col === 'price') return (p.price===null||p.price===undefined||p.price==='')?'':Number(p.price);
   393→   393→    if (col === 'oldPrice') return (p.oldPrice===null||p.oldPrice===undefined||p.oldPrice==='')?'':Number(p.oldPrice);
   394→   394→    if (col === 'featured') return (p.featured===true||p.featured==='true'||p.featured===1||p.featured==='1')?'true':'false';
   395→   395→    if (col === 'isSpecialOffer') return (p.isSpecialOffer===true||p.isSpecialOffer==='true'||p.isSpecialOffer===1||p.isSpecialOffer==='1')?'true':'false';
   396→   396→    if (col === 'stock') {
   397→   397→      // null / undefined / blank → empty cell (means unlimited on the website)
   398→   398→      if (p.stock===null||p.stock===undefined||p.stock==='') return '';
   399→   399→      var n = Number(p.stock);
   400→   400→      return isNaN(n) ? '' : n;
   401→   401→    }
   402→   402→    if (col === 'image') return coverImage;

   404→   404→    return (p[col]===undefined||p[col]===null)?'':String(p[col]);
   405→   405→  });

   519→   519→// ============================================================
   520→   520→//  STOCK DECREMENT — onEdit trigger (RESTORED)
   521→   521→// ============================================================
   522→   522→//  Watches the Orders sheet. When an order's Status changes to
   523→   523→//  "Confirmed", the matching product's Stock is decremented.
   524→   524→//
   525→   525→//  Uses Stock Key column (column P) for EXACT matching:
   526→   526→//    - If stockKey has value → find exact match in Stock tab → decrement
   527→   527→//    - If empty → try product name match → decrement
   528→   528→//    - If no match → skip (can't decrement unlimited)
   529→   529→//
   530→   530→//  To install: Apps Script editor → Triggers → Add trigger →
   531→   531→//    Function: onStockEdit
   532→   532→//    Event source: From spreadsheet
   533→   533→//    Event type: On edit
   534→   534→// ============================================================
   535→   535→
   536→   536→function onStockEdit(e) {
   537→   537→  try {
   538→   538→    var range = e && e.range;
   539→   539→    if (!range) return;
   540→   540→    var sheet = range.getSheet();
   541→   541→    if (sheet.getName() !== ORDERS_SHEET) return;
   542→   542→    if (range.getColumn() !== ORDERS_COL.STATUS + 1) return;
   543→   543→    if (range.getRow() < 2) return;
   544→   544→
   545→   545→    var newStatus = String(e.value || '').trim().toLowerCase();
   546→   546→    var oldStatus = String((e.oldValue || '')).trim().toLowerCase();
   547→   547→
   548→   548→    var row = range.getRow();
   549→   549→    var data = sheet.getRange(row, 1, 1, ORDERS_COL.STOCK_KEY + 1).getValues()[0];
   550→   550→    var productName = String(data[ORDERS_COL.PRODUCT] || '').trim();
   551→   551→    var qtyRaw = data[ORDERS_COL.QTY];
   552→   552→    var qty = (qtyRaw === '' || qtyRaw === null || qtyRaw === undefined) ? 1 : parseInt(String(qtyRaw), 10);
   553→   553→    if (isNaN(qty) || qty < 1) qty = 1;
   554→   554→    if (!productName) return;
   555→   555→
   556→   556→    // Multi-item orders (contain "+") — skip
   557→   557→    if (productName.indexOf('+') >= 0) return;
   558→   558→
   559→   559→    // Strip trailing ×N
   560→   560→    var bareName = productName.replace(/\s*[×x]\s*\d+\s*$/, '').trim();
   561→   561→    if (!bareName) return;
   562→   562→
   563→   563→    // Read Stock Key column (column P = index 15)
   564→   564→    var stockKeyStr = String(data[ORDERS_COL.STOCK_KEY] || '').trim();
   565→   565→
   566→   566→    var STOCK_DECREMENTED = ['confirmed', 'shipped', 'delivered'];
   567→   567→
   568→   568→    // CASE 1: Confirmed → DECREMENT
   569→   569→    if (STOCK_DECREMENTED.indexOf(newStatus) >= 0) {
   570→   570→      if (STOCK_DECREMENTED.indexOf(oldStatus) >= 0) return;
   571→   571→
   572→   572→      // Try stockKey FIRST (EXACT match — sent by frontend)
   573→   573→      if (stockKeyStr) {
   574→   574→        // stockKey may be comma-separated (multiple variant keys)
   575→   575→        var keys = stockKeyStr.split(',');
   576→   576→        var decremented = false;
   577→   577→        for (var k = 0; k < keys.length; k++) {
   578→   578→          var key = keys[k].trim();
   579→   579→          if (!key) continue;
   580→   580→          decremented = decrementStockByKey_(key, qty);
   581→   581→          if (decremented) break; // found and decremented
   582→   582→        }
   583→   583→        if (decremented) return; // done
   584→   584→      }
   585→   585→
   586→   586→      // Fallback: try product name match
   587→   587→      decrementProductStock_(bareName, qty);
   588→   588→    }
   589→   589→
   590→   590→    // CASE 2: Cancelled → REVERT
   591→   591→    if (newStatus === 'cancelled') {
   592→   592→      if (STOCK_DECREMENTED.indexOf(oldStatus) < 0) return;
   593→   593→
   594→   594→      if (stockKeyStr) {
   595→   595→        var keys2 = stockKeyStr.split(',');
   596→   596→        var reverted = false;
   597→   597→        for (var k2 = 0; k2 < keys2.length; k2++) {
   598→   598→          var key2 = keys2[k2].trim();
   599→   599→          if (!key2) continue;
   600→   600→          reverted = incrementStockByKey_(key2, qty);
   601→   601→          if (reverted) break;
   602→   602→        }
   603→   603→        if (reverted) return;
   604→   604→      }
   605→   605→
   606→   606→      incrementProductStock_(bareName, qty);
   607→   607→    }
   608→   608→
   609→   609→  } catch (err) {
   610→   610→    Logger.log('[onStockEdit] error: ' + err);
   611→   611→  }
   612→   612→}

   701→   701→// ============================================================
   702→   702→//  DEDUPE + CLEANUP — one-time maintenance actions
   703→   703→// ============================================================
   704→   704→
   705→   705→/**
   706→   706→ * Remove duplicate product rows (same ID appearing multiple times).
   707→   707→ * Keeps the FIRST occurrence (which is usually the original, most-complete row)
   708→   708→ * and deletes subsequent duplicates.
   709→   709→ *
   710→   710→ * Also fixes the "Meubes" → "Meubles" typo in-place for all rows.
   711→   711→ *
   712→   712→ * Called via: ?action=dedupe
   713→   713→ * Returns: { ok: true, removed: <count>, fixed_categories: <count>, remaining: <count> }
   714→   714→ */
   715→   715→function doDedupeProducts() {
   716→   716→  var sheet = ensureProductsSheet();
   717→   717→  var values = sheet.getDataRange().getValues();
   718→   718→  if (values.length < 2) return jsonOut({ ok: true, removed: 0, remaining: 0 });
   719→   719→
   720→   720→  var header = values[0];
   721→   721→  var idCol = 0;            // column A
   722→   722→  var categoryCol = header.indexOf('category');
   723→   723→  var nameCol = header.indexOf('name');
   724→   724→
   725→   725→  // First pass: fix "Meubes" → "Meubles" in-place for every row
   726→   726→  var categoryFixes = 0;
   727→   727→  if (categoryCol >= 0) {
   728→   728→    for (var i = 1; i < values.length; i++) {
   729→   729→      var cat = String(values[i][categoryCol] || '').trim();
   730→   730→      if (cat === 'Meubes') {
   731→   731→        sheet.getRange(i + 1, categoryCol + 1).setValue('Meubles');
   732→   732→        categoryFixes++;
   733→   733→      }
   734→   734→    }
   735→   735→  }
   736→   736→
   737→   737→  // Second pass: find duplicate IDs and mark rows for deletion
   738→   738→  // (iterate from the END so deleting doesn't shift indices we still need)
   739→   739→  var seenIds = {};
   740→   740→  var rowsToDelete = [];
   741→   741→  for (var j = 1; j < values.length; j++) {
   742→   742→    var idVal = String(values[j][idCol] || '').trim();
   743→   743→    if (!idVal) continue;
   744→   744→    // Skip guidance rows (emoji/Arabic in ID)
   745→   745→    if (/[\u0600-\u06FF\u{1F000}-\u{1FFFF}]/u.test(idVal)) continue;
   746→   746→    if (seenIds[idVal]) {
   747→   747→      rowsToDelete.push(j + 1); // sheet rows are 1-indexed, +1 because header is row 1
   748→   748→    } else {
   749→   749→      seenIds[idVal] = true;
   750→   750→    }
   751→   751→  }
   752→   752→
   753→   753→  // Delete rows from the BOTTOM UP so indices stay valid
   754→   754→  rowsToDelete.sort(function(a, b) { return b - a; });
   755→   755→  for (var k = 0; k < rowsToDelete.length; k++) {
   756→   756→    sheet.deleteRow(rowsToDelete[k]);
   757→   757→  }
   758→   758→
   759→   759→  var remaining = sheet.getLastRow() - 1; // minus header
   760→   760→  return jsonOut({
   761→   761→    ok: true,
   762→   762→    removed: rowsToDelete.length,
   763→   763→    fixed_categories: categoryFixes,
   764→   764→    remaining: remaining
   765→   765→  });
   766→   766→}

   768→   768→/**
   769→   769→ * Full sheet cleanup — runs dedupe + fixes category typos + removes
   770→   770→ * completely empty rows (no id, no name, no image).

   772→   772→ * Called via: ?action=cleanup
