    41→// Resize image for optimal quality + storage.
    42→// Settings (optimized for Cloudflare R2 — 9500 products in 10GB):
    43→//   - maxSize: 850px (sharp on all screens, ~70KB per image)
    44→//   - quality: 0.85 (visually identical to 0.93, half the size)
    45→//   - Format: WebP (30% smaller than JPEG)
    46→//   - Max 5 photos per product
    47→//   - PNG transparency preserved when source is PNG
    48→async function resizeImage(
    49→  file: File,
    50→  maxSize = 850,
    51→  budget = 200_000,
    52→): Promise<string> {
    53→  return new Promise((resolve, reject) => {
    54→    const reader = new FileReader();
    55→    reader.onload = () => {
    56→      const img = new Image();
    57→      img.onload = () => {
    58→        let { width, height } = img;
    59→        // Only downscale if the image exceeds maxSize; never upscale.
    60→        if (width > height && width > maxSize) {
    61→          height = Math.round((height * maxSize) / width);
    62→          width = maxSize;
    63→        } else if (height > maxSize) {
    64→          width = Math.round((width * maxSize) / height);
    65→          height = maxSize;
    66→        }
    67→        const canvas = document.createElement("canvas");
    68→        canvas.width = width;
    69→        canvas.height = height;
    70→        const ctx = canvas.getContext("2d");
    71→        if (!ctx) {
    72→          reject(new Error("canvas not supported"));
    73→          return;
    74→        }
    75→        // Maximum-quality smoothing
    76→        ctx.imageSmoothingEnabled = true;
    77→        ctx.imageSmoothingQuality = "high";
    78→        ctx.drawImage(img, 0, 0, width, height);
    79→
    80→        // Use WebP for all images (30% smaller than JPEG, supported everywhere).
    81→        // Preserve PNG transparency by keeping PNG format for PNG sources.
    82→        const isPng = file.type === "image/png";
    83→        let quality = 0.85;
    84→        let mimeType = isPng ? "image/png" : "image/webp";
    85→        let dataUrl = canvas.toDataURL(mimeType, quality);
    86→
    87→        // Fallback: progressively reduce quality until it fits within the budget.
    88→        if (dataUrl.length > budget) {
    89→          mimeType = "image/webp";
    90→          quality = 0.80;
    91→          dataUrl = canvas.toDataURL(mimeType, quality);
    92→          while (dataUrl.length > budget && quality > 0.50) {
    93→            quality -= 0.05;
    94→            dataUrl = canvas.toDataURL(mimeType, quality);
    95→          }
    96→          // If still too large, reduce resolution
    97→          while (dataUrl.length > budget && width > 400) {
    98→            width = Math.round(width * 0.85);
    99→            height = Math.round(height * 0.85);
   100→            canvas.width = width;
   101→            canvas.height = height;
   102→            ctx.drawImage(img, 0, 0, width, height);
   103→            dataUrl = canvas.toDataURL(mimeType, quality);
   104→          }
   105→        }
   106→        resolve(dataUrl);
   107→      };
   108→      img.onerror = () => reject(new Error("image load failed"));
   109→      img.src = reader.result as string;
   110→    };
   111→    reader.onerror = () => reject(new Error("file read failed"));
   112→    reader.readAsDataURL(file);
   113→  });
   114→}

   116→function PasswordGate({ onAuthed }: { onAuthed: () => void }) {
   117→  const [value, setValue] = useState("");
   118→  const [error, setError] = useState(false);
   119→
   120→  const submit = () => {
   121→    if (value === ADMIN_PASSWORD) {
   122→      try {
   123→        sessionStorage.setItem(SESSION_KEY, "1");
   124→      } catch {
   125→        // ignore
   126→      }
   127→      onAuthed();
   128→    } else {
   129→      setError(true);
   130→    }
   131→  };
   132→
   133→  return (
   134→    <div
   135→      className="flex min-h-[60vh] items-center justify-center p-6"
   136→     
   137→     
   138→    >
   139→      <div className="w-full max-w-sm rounded-2xl border border-emerald/30 bg-night-soft/80 p-6 shadow-2xl backdrop-blur-md">
   140→        <div className="mb-4 flex flex-col items-center text-center">
   141→          <div className="mb-3 flex h-12 w-12 items-center justify-center rounded-full bg-emerald/10">
   142→            <Lock className="h-6 w-6 text-emerald" />
   143→          </div>
   144→          <h2 className="font-arabic text-2xl font-bold text-charcoal">
   145→            لوحة التحكم
   146→          </h2>
   147→          <p className="mt-1 font-arabic text-sm text-gray">
   148→            أدخل كلمة المرور للوصول إلى لوحة إدارة المنتجات.
   149→          </p>
   150→        </div>
   151→        <input
   152→          type="password"
   153→          value={value}
   154→          onChange={(e) => {
   155→            setValue(e.target.value);
   156→            setError(false);
   157→          }}
   158→          onKeyDown={(e) => {
   159→            if (e.key === "Enter") submit();
   160→          }}
   161→          className={`w-full rounded-lg border bg-night/60 px-3 py-2.5 text-left font-arabic text-sm text-charcoal outline-none focus:border-emerald ${
   162→            error ? "border-terracotta" : "border-clay/40"
   163→          }`}
   164→          placeholder="كلمة المرور"
   165→          autoFocus
   166→        />
   167→        {error && (
   168→          <p className="mt-2 font-arabic text-xs text-terracotta">
   169→            كلمة المرور غير صحيحة.
   170→          </p>
   171→        )}
   172→        <button
   173→          type="button"
   174→          onClick={submit}
   175→          className="mt-4 w-full rounded-full bg-emerald px-4 py-2.5 text-sm font-semibold text-night hover:bg-emerald-bright"
   176→          style={{ boxShadow: "0 0 18px rgba(42, 125, 91, 0.45)" }}
   177→        >
   178→          دخول
   179→        </button>
   180→      </div>
   181→    </div>
   182→  );
   183→}

   185→function EditForm({
   186→  product,
   187→  categories,
   188→  onSave,
   189→  onCancel,
   190→}: {
   191→  product: Product;
   192→  categories: string[];
   193→  onSave: (p: Product) => void;
   194→  onCancel: () => void;
   195→}) {
   196→  const [draft, setDraft] = useState<Product>(product);
   197→  const [uploading, setUploading] = useState(false);
   198→  const [dragOver, setDragOver] = useState(false);
   199→  const fileRef = useRef<HTMLInputElement>(null);
   200→
   201→  useEffect(() => {
   202→    setDraft(product);
   203→  }, [product]);
   204→
   205→  const photos = getProductImages(draft);
   206→  // Allow up to 5 high-quality photos per product.
   207→  const MAX_PHOTOS = 8;
   208→
   209→  const syncPhotos = (next: string[]) => {
   210→    const cleaned = next.filter((s) => s && s.trim() !== "");
   211→    setDraft((d) => ({
   212→      ...d,
   213→      images: cleaned,
   214→      image: cleaned[0] ?? "",
   215→    }));
   216→  };
   217→
   218→  const handleFiles = async (files: FileList | File[]) => {
   219→    const arr = Array.from(files).filter((f) => f.type.startsWith("image/"));
   220→    if (arr.length === 0) {
   221→      toast.error("الرجاء اختيار ملف صورة.");
   222→      return;
   223→    }
   224→    const remaining = MAX_PHOTOS - photos.length;
   225→    if (remaining <= 0) {
   226→      toast.error(`الحد الأقصى ${MAX_PHOTOS} صور لكل منتج.`);
   227→      return;
   228→    }
   229→    const toProcess = arr.slice(0, remaining);
   230→    if (arr.length > remaining) {
   231→      toast.message(`سيتم إضافة أول ${remaining} صور فقط.`);
   232→    }
   233→    setUploading(true);
   234→    try {
   235→      const newPhotos: string[] = [];
   236→      for (const f of toProcess) {
   237→        try {
   238→          const dataUrl = await resizeImage(f, 850, 200_000);
   239→          newPhotos.push(dataUrl);
   240→        } catch {
   241→          // skip failed image
   242→        }
   243→      }
   244→      if (newPhotos.length === 0) {
   245→        toast.error("فشل في معالجة الصور.");
   246→        return;
   247→      }
   248→      syncPhotos([...photos, ...newPhotos]);
   249→      toast.success(
   250→        `تمت إضافة ${newPhotos.length} صورة${newPhotos.length > 1 ? " بنجاح" : " بنجاح"}.`,
   251→      );
   252→    } finally {
   253→      setUploading(false);
   254→    }
   255→  };
   256→
   257→  const onDrop = (e: React.DragEvent) => {
   258→    e.preventDefault();
   259→    setDragOver(false);
   260→    const files = e.dataTransfer.files;
   261→    if (files && files.length > 0) handleFiles(files);
   262→  };
   263→
   264→  const removePhoto = (idx: number) => {
   265→    const next = [...photos];
   266→    next.splice(idx, 1);
   267→    syncPhotos(next);
   268→  };
   269→
   270→  const setCoverPhoto = (idx: number) => {
   271→    if (idx === 0 || idx >= photos.length) return;
   272→    const next = [...photos];
   273→    const [picked] = next.splice(idx, 1);
   274→    next.unshift(picked);
   275→    syncPhotos(next);
   276→  };
   277→
   278→  // Variants management — simple Colors + Sizes sections (replaces the old
   279→  // generic "variations" editor). Each entry has a name + optional price
   280→  // adjustment. Both lists live on the same `draft.variants` array, tagged
   281→  // with `type: "color" | "size"`. The two sections are rendered by filtering
   282→  // the same array; the GLOBAL array index is passed to each handler so the
   283→  // state update maps over the right entry.
   284→  const variants: ProductVariant[] = draft.variants ?? [];
   285→
   286→  const addVariant = (type: "color" | "size") => {
   287→    setDraft((d) => ({
   288→      ...d,
   289→      variants: [...(d.variants ?? []), { type, name: "", priceAdjustment: 0 }],
   290→    }));
   291→  };
   292→  const removeVariant = (idx: number) => {
   293→    setDraft((d) => ({
   294→      ...d,
   295→      variants: (d.variants ?? []).filter((_, i) => i !== idx),
   296→    }));
   297→  };
   298→  const updateVariantName = (idx: number, name: string) => {
   299→    setDraft((d) => ({
   300→      ...d,
   301→      variants: (d.variants ?? []).map((v, i) =>
   302→        i === idx ? { ...v, name } : v,
   303→      ),
   304→    }));
   305→  };
   306→  const updateVariantAdjustment = (idx: number, adj: number) => {
   307→    setDraft((d) => ({
   308→      ...d,
   309→      variants: (d.variants ?? []).map((v, i) =>
   310→        i === idx ? { ...v, priceAdjustment: isNaN(adj) ? 0 : adj } : v,
   311→      ),
   312→    }));
   313→  };
   314→
   315→  // Quantity tiers management — special offers at specific quantities.
   316→  // Each tier can combine a free-shipping benefit AND a discount amount.
   317→  const tiers = draft.quantityTiers ?? [];
   318→
   319→  const addTier = () => {
   320→    const newTiers: QuantityTier[] = [
   321→      ...tiers,
   322→      { qty: 2, freeShipping: "both" },
   323→    ];
   324→    setDraft({ ...draft, quantityTiers: newTiers });
   325→  };
   326→
   327→  const removeTier = (idx: number) => {
   328→    const newTiers = tiers.filter((_, i) => i !== idx);
   329→    setDraft({ ...draft, quantityTiers: newTiers });
   330→  };
   331→
   332→  const updateTier = (
   333→    idx: number,
   334→    field: keyof QuantityTier,
   335→    value: number | QuantityTier["freeShipping"],
   336→  ) => {
   337→    const newTiers = [...tiers];
   338→    const prev = newTiers[idx];
   339→    if (field === "qty") {
   340→      newTiers[idx] = { ...prev, qty: Number(value) };
   341→    } else if (field === "freeShipping") {
   342→      newTiers[idx] = {
   343→        ...prev,
   344→        freeShipping: value as QuantityTier["freeShipping"],
   345→      };
   346→    } else if (field === "discountAmount") {
   347→      const n = Number(value);
   348→      newTiers[idx] = {
   349→        ...prev,
   350→        discountAmount: n > 0 ? n : undefined,
   351→      };
   352→    }
   353→    setDraft({ ...draft, quantityTiers: newTiers });
   354→  };
   355→
   356→  const save = () => {
   357→    const nameStr = String(draft.name ?? "").trim();
   358→    if (!nameStr) {
   359→      toast.error("الاسم مطلوب.");
   360→      return;
   361→    }
   362→    onSave(draft);
   363→  };
   364→
   365→  const inputClass =
   366→    "w-full rounded-lg border border-clay/40 bg-night/60 px-3 py-2.5 text-left font-arabic text-sm text-charcoal outline-none focus:border-brass focus:ring-1 focus:ring-brass/30";
   367→
   368→  return (
   369→    <div
   370→      className="rounded-2xl border border-emerald/25 bg-night-soft/70 p-5 backdrop-blur-md sm:p-6"
   371→     
   372→     
   373→    >
   374→      <div className="mb-4 flex items-center justify-between">
   375→        <h3 className="font-arabic text-xl font-bold text-charcoal">
   376→          {product.name ? "تعديل المنتج" : "منتج جديد"}
   377→        </h3>
   378→        <button
   379→          type="button"
   380→          onClick={onCancel}
   381→          aria-label="إلغاء"
   382→          className="flex h-8 w-8 items-center justify-center rounded-full text-charcoal hover:bg-emerald/10"
   383→        >
   384→          <X className="h-5 w-5" />
   385→        </button>
   386→      </div>
   387→
   388→      <div className="space-y-5 font-arabic">
   389→        {/* 1. Multi-photo upload */}
   390→        <div>
   391→          <div className="mb-1 flex items-center justify-between">
   392→            <label className="block text-sm font-medium text-charcoal">
   393→              الصور{" "}
   394→              <span className="text-xs font-normal text-gray-light">
   395→                (الأولى = الغلاف · الحد الأقصى {MAX_PHOTOS})
   396→              </span>
   397→            </label>
   398→            <span
   399→              className={`text-[11px] font-medium ${
   400→                photos.length >= MAX_PHOTOS ? "text-terracotta" : "text-gray-light"
   401→              }`}
   402→            >
   403→              {photos.length} / {MAX_PHOTOS}
   404→            </span>
   405→          </div>
   406→
   407→          {/* Existing photos grid */}
   408→          {photos.length > 0 && (
   409→            <div className="mb-2 grid grid-cols-3 gap-2 sm:grid-cols-4">
   410→              {photos.map((p, i) => (
   411→                <div
   412→                  key={i}
   413→                  className="group relative aspect-square overflow-hidden rounded-lg border border-clay/40 bg-night"
   414→                >
   415→                  <img
   416→                    src={p}
   417→                    alt={`صورة ${i + 1}`}
   418→                    className="h-full w-full object-contain"
   419→                  />
   420→                  {i === 0 && (
   421→                    <span className="absolute right-1 top-1 rounded-full bg-brass/90 px-1.5 py-0.5 text-[9px] font-semibold text-white">
   422→                      الغلاف
   423→                    </span>
   424→                  )}
   425→                  <div className="absolute inset-0 flex items-center justify-center gap-1 bg-ink/60 opacity-0 transition-opacity group-hover:opacity-100">
   426→                    {i !== 0 && (
   427→                      <button
   428→                        type="button"
   429→                        onClick={() => setCoverPhoto(i)}
   430→                        aria-label="تعيين كغلاف"
   431→                        title="تعيين كغلاف"
   432→                        className="flex h-7 w-7 items-center justify-center rounded-full bg-night/90 text-emerald hover:bg-night-soft border border-emerald/30"
   433→                      >
   434→                        <Star className="h-3.5 w-3.5" fill="currentColor" />
   435→                      </button>
   436→                    )}
   437→                    <button
   438→                      type="button"
   439→                      onClick={() => removePhoto(i)}
   440→                      aria-label="حذف هذه الصورة"
   441→                      title="حذف"
   442→                      className="flex h-7 w-7 items-center justify-center rounded-full bg-night/90 text-terracotta hover:bg-night-soft border border-terracotta/30"
   443→                    >
   444→                      <Trash2 className="h-3.5 w-3.5" />
   445→                    </button>
   446→                  </div>
   447→                </div>
   448→              ))}
   449→            </div>
   450→          )}
   451→
   452→          {/* Drop zone */}
   453→          <div
   454→            onClick={() => !uploading && fileRef.current?.click()}
   455→            onDragOver={(e) => {
   456→              if (uploading) return;
   457→              e.preventDefault();
   458→              setDragOver(true);
   459→            }}
   460→            onDragLeave={() => setDragOver(false)}
   461→            onDrop={(e) => { if (!uploading) onDrop(e); }}
   462→            className={`flex flex-col items-center justify-center gap-1.5 rounded-xl border-2 border-dashed p-4 text-center transition-colors ${
   463→              uploading
   464→                ? "border-clay/40 bg-night/20 cursor-wait pointer-events-none"
   465→                : "cursor-pointer " + (dragOver
   466→                  ? "border-emerald bg-emerald/5"
   467→                  : "border-clay/40 bg-night/30 hover:border-emerald/50")
   468→            }`}
   469→          >
   470→            <div className="flex h-12 w-12 flex-col items-center justify-center text-gray-light">
   471→              {uploading ? (
   472→                <span className="text-xs text-emerald">جاري المعالجة...</span>
   473→              ) : (
   474→                <ImageIcon className="h-6 w-6" />
   475→              )}
   476→            </div>
   477→            <span className="flex items-center gap-1 text-xs font-medium text-emerald">
   478→              <Upload className="h-3.5 w-3.5" />
   479→              {uploading
   480→                ? "جاري المعالجة..."
   481→                : photos.length === 0
   482→                  ? "إضافة صور"
   483→                  : "إضافة المزيد من الصور"}
   484→            </span>
   485→            <span className="text-[10px] text-gray-light">
   486→              انقر أو اسحب وأفلت الصور هنا
   487→            </span>
   488→          </div>
   489→          <input
   490→            ref={fileRef}
   491→            type="file"
   492→            accept="image/*"
   493→            multiple
   494→            className="hidden"
   495→            onChange={(e) => {
   496→              if (e.target.files && e.target.files.length > 0) {
   497→                handleFiles(e.target.files);
   498→              }
   499→              e.target.value = "";
   500→            }}
   501→          />
   502→        </div>
   503→
   504→        {/* ===== Section: المعلومات الأساسية ===== */}
   505→        <h4 className="admin-section-title">المعلومات الأساسية</h4>
   506→
   507→        {/* 2. Name */}
   508→        <div>
   509→          <label htmlFor={`fld-name-${draft.id ?? "new"}`} className="mb-1 block text-sm font-medium text-charcoal">
   510→            الاسم <span className="text-terracotta">*</span>
   511→          </label>
   512→          <input
   513→            id={`fld-name-${draft.id ?? "new"}`}
   514→            name="productName"
   515→            type="text"
   516→            value={draft.name}
   517→            onChange={(e) => setDraft({ ...draft, name: e.target.value })}
   518→            className={inputClass}
   519→            placeholder="اسم المنتج"
   520→            autoComplete="off"
   521→            autoCorrect="off"
   522→            spellCheck={false}
   523→          />
   524→        </div>
   525→
   526→        {/* 3. Description */}
   527→        <div>
   528→          <label htmlFor={`fld-desc-${draft.id ?? "new"}`} className="mb-1 block text-sm font-medium text-charcoal">
   529→            الوصف
   530→          </label>
   531→          <textarea
   532→            id={`fld-desc-${draft.id ?? "new"}`}
   533→            name="productDescription"
   534→            value={draft.description}
   535→            onChange={(e) =>
   536→              setDraft({ ...draft, description: e.target.value })
   537→            }
   538→            rows={5}
   539→            className={`${inputClass} resize-none`}
   540→            placeholder="وصف المنتج بالعربية..."
   541→            autoComplete="off"
   542→            autoCorrect="off"
   543→            spellCheck={false}
   544→          />
   545→        </div>
   546→
   547→        {/* 4. Category */}
   548→        <div>
   549→          <label htmlFor={`fld-category-${draft.id ?? "new"}`} className="mb-1 block text-sm font-medium text-charcoal">
   550→            الفئة{" "}
   551→            <span className="text-xs font-normal text-gray-light">
   552→              (اختياري)
   553→            </span>
   554→          </label>
   555→          <input
   556→            id={`fld-category-${draft.id ?? "new"}`}
   557→            name="productCategory"
   558→            type="text"
   559→            value={draft.category}
   560→            onChange={(e) =>
   561→              setDraft({ ...draft, category: e.target.value })
   562→            }
   563→            className={inputClass}
   564→            placeholder="مثال: إكسسوارات السيارة"
   565→            list="rokn-category-list"
   566→            autoComplete="off"
   567→            autoCorrect="off"
   568→            spellCheck={false}
