     1→"use client";
     2→
     3→import { useEffect, useMemo, useState } from "react";
     4→import { toast } from "sonner";
     5→import { Check, Send, Heart, Sparkles } from "lucide-react";
     6→import {
     7→  getShippingPrice,
     8→  ShippingSpeed,
     9→  DeliveryType,
    10→  SHIPPING_SPEED_LABELS_AR,
    11→  DELIVERY_TYPE_LABELS_AR,
    12→} from "@/lib/shipping";
    13→import { formatPrice, QuantityTier } from "@/lib/products";
    14→import { useAlgeriaData, PHONE_REGEX } from "@/hooks/use-algeria-data";
    15→import { BRAND } from "@/lib/brand-config";
    16→
    17→export type OrderItem = {
    18→  name: string;
    19→  price: number | null;
    20→  quantity: number;
    21→};
    22→
    23→type CodOrderFormProps = {
    24→  items: OrderItem[];
    25→  rupture?: boolean;
    26→  onSuccess?: () => void;
    27→  onContinueShopping?: () => void;
    28→  /** Extra notes that should be appended to the order notes (e.g. variation selections). */
    29→  extraNotes?: string;
    30→  /** Optional quantity tiers — when provided AND items.length === 1, the 4-button
    31→   *  selector highlights tier quantities with a ✨ badge and applies discount /
    32→   *  free-shipping benefits to the totals. */
    33→  quantityTiers?: QuantityTier[];
    34→};
    35→
    36→type FormState = {
    37→  fullName: string;
    38→  phone: string;
    39→  wilaya: string;
    40→  commune: string;
    41→  speed: ShippingSpeed;
    42→  delivery: DeliveryType;
    43→  notes: string;
    44→};
    45→
    46→function generateOrderRef(): string {
    47→  const n = Math.floor(100000 + Math.random() * 900000);
    48→  return `SD-${n}`;
    49→}
    50→
    51→export function CodOrderForm({
    52→  items: initialItems,
    53→  rupture,
    54→  onSuccess,
    55→  onContinueShopping,
    56→  extraNotes,
    57→  quantityTiers,
    58→}: CodOrderFormProps) {
    59→  const { wilayas, getCommunesForWilaya, loading } = useAlgeriaData();
    60→  const [items, setItems] = useState<OrderItem[]>(initialItems);
    61→  const [form, setForm] = useState<FormState>({
    62→    fullName: "",
    63→    phone: "",
    64→    wilaya: "",
    65→    commune: "",
    66→    speed: "express",
    67→    delivery: "stop_desk",
    68→    notes: "",
    69→  });
    70→  const [phoneError, setPhoneError] = useState(false);
    71→  const [submitting, setSubmitting] = useState(false);
    72→  const [done, setDone] = useState(false);
    73→  const [orderRef, setOrderRef] = useState("");
    74→  const [orderSummary, setOrderSummary] = useState<null | {
    75→    items: OrderItem[];
    76→    productTotal: number;
    77→    discountAmount: number;
    78→    shippingPrice: number;
    79→    grandTotal: number;
    80→    freeShippingApplied: boolean;
    81→    fullName: string;
    82→    phone: string;
    83→    wilayaLabel: string;
    84→    deliveryLabel: string;
    85→    date: string;
    86→  }>(null);
    87→
    88→  // Sync items when prop changes (e.g., modal opens different product)
    89→  useEffect(() => {
    90→    setItems(initialItems);
    91→  }, [initialItems]);
    92→
    93→  const communes = useMemo(
    94→    () => (form.wilaya ? getCommunesForWilaya(form.wilaya) : []),
    95→    [form.wilaya, getCommunesForWilaya],
    96→  );
    97→
    98→  // Reset commune when wilaya changes
    99→  useEffect(() => {
   100→    setForm((f) => ({ ...f, commune: "" }));
   101→  }, [form.wilaya]);
   102→
   103→  // The NORMAL shipping price (without any tier benefit applied).
   104→  // Returns null until BOTH wilaya AND commune are selected.
   105→  const normalShippingPrice = useMemo(() => {
   106→    if (!form.wilaya || !form.commune) return null;
   107→    const result = getShippingPrice(form.wilaya, form.speed, form.delivery);
   108→    return result ? result.price : null;
   109→  }, [form.wilaya, form.commune, form.speed, form.delivery]);
   110→
   111→  const shippingDelay = useMemo(() => {
   112→    if (!form.wilaya || !form.commune) return null;
   113→    const result = getShippingPrice(form.wilaya, form.speed, form.delivery);
   114→    return result ? result.delay : null;
   115→  }, [form.wilaya, form.commune, form.speed, form.delivery]);
   116→
   117→  // Active tier — only for single-item orders where the parent passed `quantityTiers`.
   118→  // Used to compute discount and free-shipping benefits.
   119→  const activeTier = useMemo(() => {
   120→    if (items.length !== 1) return null;
   121→    const q = items[0].quantity;
   122→    return (quantityTiers ?? []).find((t) => t.qty === q) ?? null;
   123→  }, [items, quantityTiers]);
   124→
   125→  // Discount from the active tier (if any) — clamped so total can never go negative.
   126→  const discountAmount = activeTier?.discountAmount ?? 0;
   127→
   128→  // Whether the active tier grants free shipping for the current delivery type.
   129→  const freeShippingApplied = useMemo(() => {
   130→    if (!activeTier || activeTier.freeShipping === "none") return false;
   131→    if (activeTier.freeShipping === "both") return true;
   132→    if (activeTier.freeShipping === "desk" && form.delivery === "stop_desk") return true;
   133→    if (activeTier.freeShipping === "home" && form.delivery === "home") return true;
   134→    return false;
   135→  }, [activeTier, form.delivery]);
   136→
   137→  const productTotal = useMemo(
   138→    () => items.reduce((s, i) => s + (i.price ?? 0) * i.quantity, 0),
   139→    [items],
   140→  );
   141→
   142→  // Product total AFTER discount (never negative).
   143→  const productTotalAfterDiscount = useMemo(
   144→    () => Math.max(0, productTotal - discountAmount),
   145→    [productTotal, discountAmount],
   146→  );
   147→
   148→  // Final shipping price: 0 if the tier grants free shipping, otherwise normal price.
   149→  const shippingPrice = useMemo(() => {
   150→    if (normalShippingPrice === null) return null;
   151→    return freeShippingApplied ? 0 : normalShippingPrice;
   152→  }, [normalShippingPrice, freeShippingApplied]);
   153→
   154→  const grandTotal = useMemo(() => {
   155→    const ship = shippingPrice ?? 0;
   156→    return productTotalAfterDiscount + ship;
   157→  }, [productTotalAfterDiscount, shippingPrice]);
   158→
   159→  const set = <K extends keyof FormState>(key: K, value: FormState[K]) => {
   160→    setForm((f) => ({ ...f, [key]: value }));
   161→    if (key === "phone") setPhoneError(false);
   162→  };
   163→
   164→  const setSingleQty = (q: number) => {
   165→    if (q < 1) return;
   166→    setItems((prev) =>
   167→      prev.length === 1 ? [{ ...prev[0], quantity: q }] : prev,
   168→    );
   169→  };
   170→
   171→  const tierBenefitLine = (tier: QuantityTier): string => {
   172→    const hasFree = tier.freeShipping && tier.freeShipping !== "none";
   173→    const discount = tier.discountAmount ?? 0;
   174→    const hasDiscount = discount > 0;
   175→    const shippingLabel =
   176→      tier.freeShipping === "both"
   177→        ? "كلاهما"
   178→        : tier.freeShipping === "desk"
   179→        ? "المكتب"
   180→        : tier.freeShipping === "home"
   181→        ? "المنزل"
   182→        : "";
   183→    if (hasFree && hasDiscount) {
   184→      return `🎉 توصيل مجاني (${shippingLabel}) + خصم ${discount} دج عند شراء ${tier.qty} قطعة!`;
   185→    }
   186→    if (hasFree) {
   187→      return `🎉 توصيل مجاني (${shippingLabel}) عند شراء ${tier.qty} قطعة!`;
   188→    }
   189→    if (hasDiscount) {
   190→      return `🎉 خصم ${discount} دج عند شراء ${tier.qty} قطعة!`;
   191→    }
   192→    return "";
   193→  };
   194→
   195→  const validate = (): boolean => {
   196→    if (!form.fullName.trim()) {
   197→      toast.error("الرجاء إدخال الاسم الكامل");
   198→      return false;
   199→    }
   200→    if (!PHONE_REGEX.test(form.phone.replace(/\D/g, ""))) {
   201→      setPhoneError(true);
   202→      toast.error(
   203→        "رقم الهاتف غير صحيح. يجب أن يبدأ بـ 05 أو 06 أو 07 ويتكون من 10 أرقام",
   204→      );
   205→      return false;
   206→    }
   207→    if (!form.wilaya) {
   208→      toast.error("الرجاء اختيار الولاية");
   209→      return false;
   210→    }
   211→    if (!form.commune) {
   212→      toast.error("الرجاء اختيار البلدية");
   213→      return false;
   214→    }
   215→    return true;
   216→  };
   217→
   218→  const handleSubmit = async () => {
   219→    if (rupture) return;
   220→    if (!validate()) return;
   221→    setSubmitting(true);
   222→
   223→    const wilayaLabel =
   224→      wilayas.find((w) => w.code === form.wilaya)?.name || form.wilaya;
   225→    const deliveryLabel = DELIVERY_TYPE_LABELS_AR[form.delivery];
   226→    const companyLabel = SHIPPING_SPEED_LABELS_AR[form.speed];
   227→    const ship = shippingPrice ?? 0;
   228→    const finalGrandTotal = productTotalAfterDiscount + ship;
   229→
   230→    try {
   231→      // Send ALL items as ONE single order (1 row in the sheet, 1 shipping price)
   232→      const allProducts = items
   233→        .map((it) => `${it.name} ×${it.quantity}`)
   234→        .join(" + ");
   235→      const totalQty = items.reduce((s, i) => s + i.quantity, 0);
   236→      const totalUnitPrice = items.reduce((s, i) => s + (i.price ?? 0), 0);
   237→
   238→      // Combine user notes with variation summary (if any)
   239→      const speedNote = shippingDelay !== null
   240→        ? `${SHIPPING_SPEED_LABELS_AR[form.speed]} — ${shippingDelay} ${shippingDelay === 1 ? "يوم" : "أيام"}`
   241→        : SHIPPING_SPEED_LABELS_AR[form.speed];
   242→
   243→      const combinedNotes = [extraNotes, form.notes, speedNote]
   244→        .filter((s) => s && s.trim())
   245→        .join(" · ");
   246→
   247→      // If a discount or free shipping was applied, append a small note so the
   248→      // store owner can see the benefit granted to this customer.
   249→      const benefitNote =
   250→        discountAmount > 0 && freeShippingApplied
   251→          ? `🎉 خصم ${discountAmount} دج + توصيل مجاني`
   252→          : discountAmount > 0
   253→          ? `🎉 خصم ${discountAmount} دج`
   254→          : freeShippingApplied
   255→          ? `🎉 توصيل مجاني`
   256→          : "";
   257→      const finalNotes = [combinedNotes, benefitNote]
   258→        .filter(Boolean)
   259→        .join(" · ");
   260→
   261→      const res = await fetch("/api/order", {
   262→        method: "POST",
   263→        headers: { "Content-Type": "application/json" },
   264→        body: JSON.stringify({
   265→          product: allProducts,
   266→          quantity: String(totalQty),
   267→          price: items.length === 1 ? items[0].price : totalUnitPrice,
   268→          total: productTotalAfterDiscount,
   269→          shippingPrice: ship,
   270→          grandTotal: finalGrandTotal,
   271→          shippingCompany: form.speed,
   272→          shippingCompanyLabel: companyLabel,
   273→          fullName: form.fullName,
   274→          phone: form.phone.replace(/\D/g, ""),
   275→          wilaya: wilayaLabel,
   276→          commune: form.commune,
   277→          delivery: form.delivery,
   278→          deliveryLabel,
   279→          notes: finalNotes,
   280→        }),
   281→      });
   282→
   283→      // If the API returned an error, don't show the thank-you screen
   284→      if (!res.ok) {
   285→        throw new Error("Order submission failed");
   286→      }
   287→
   288→      setOrderRef(generateOrderRef());
   289→      setOrderSummary({
   290→        items,
   291→        productTotal,
   292→        discountAmount,
   293→        shippingPrice: ship,
   294→        grandTotal: finalGrandTotal,
   295→        freeShippingApplied,
   296→        fullName: form.fullName,
   297→        phone: form.phone,
   298→        wilayaLabel,
   299→        deliveryLabel,
   300→        date: new Date().toLocaleString("fr-FR"),
   301→      });
   302→      setDone(true);
   303→      onSuccess?.();
   304→    } catch {
   305→      toast.error("حدث خطأ أثناء إرسال الطلب. حاول مرة أخرى.");
   306→    } finally {
   307→      setSubmitting(false);
   308→    }
   309→  };
   310→
   311→  // === RUPTURE STATE ===
   312→  if (rupture) {
   313→    return (
   314→      <div
   315→        dir="rtl"
   316→        lang="ar"
   317→        className="rounded-xl border border-terracotta/40 bg-terracotta/10 p-6 text-center font-arabic backdrop-blur-sm"
   318→      >
   319→        <p className="text-lg font-semibold text-terracotta neon-text-magenta">نفدت الكمية</p>
   320→        <p className="mt-2 text-[13px] leading-relaxed text-gray">
   321→          هذا المنتج غير متوفر حالياً. تواصل معنا لمعرفة موعد توفره القادم.
   322→        </p>
   323→        <div className="mt-5 flex flex-col gap-2">
   324→          <button
   325→            type="button"
   326→            onClick={() => onContinueShopping?.()}
   327→            className="w-full rounded-full bg-emerald px-4 py-3 text-sm font-semibold text-night transition-colors hover:bg-emerald-bright"
   328→            style={{ boxShadow: "0 0 18px rgba(42, 125, 91, 0.45)" }}
   329→          >
   330→            متابعة التسوق
   331→          </button>
   332→        </div>
   333→      </div>
   334→    );
   335→  }
   336→
   337→  // === THANK-YOU STATE — super elegant, beautiful, with love ===
   338→  if (done && orderSummary) {
   339→    return (
   340→      <div
   341→        dir="rtl"
   342→        lang="ar"
   343→        className="px-1 py-2 text-center font-arabic"
   344→      >
   345→        {/* Animated heart with neon glow + floating sparkles */}
   346→        <div className="relative mx-auto mb-5 h-24 w-24">
   347→          {/* Outer pulsing glow rings */}
   348→          <div
   349→            aria-hidden
   350→            className="pulse-soft absolute inset-0 rounded-full"
   351→            style={{
   352→              background:
   353→                "radial-gradient(circle, rgba(194, 91, 126, 0.25) 0%, transparent 70%)",
   354→              filter: "blur(12px)",
   355→            }}
   356→          />
   357→          <div
   358→            aria-hidden
   359→            className="pulse-soft absolute inset-0 rounded-full"
   360→            style={{
   361→              background:
   362→                "radial-gradient(circle, rgba(15, 13, 10, 0.22) 0%, transparent 70%)",
   363→              filter: "blur(16px)",
   364→              animationDelay: "0.5s",
   365→            }}
   366→          />
   367→
   368→          {/* Floating sparkle stars around the heart */}
   369→          <div
   370→            className="float-strong absolute -top-1 -right-1 text-brass-bright"
   371→          >
   372→            <Sparkles className="h-4 w-4" fill="currentColor" />
   373→          </div>
   374→          <div
   375→            className="float-strong absolute -bottom-1 -left-2 text-neon-magenta"
   376→            style={{ animationDelay: "0.3s" }}
   377→          >
   378→            <Sparkles className="h-3.5 w-3.5" fill="currentColor" />
   379→          </div>
   380→          <div
   381→            className="float-strong absolute top-1/2 -right-3 text-charcoal"
   382→            style={{ animationDelay: "0.7s" }}
   383→          >
   384→            <Sparkles className="h-3 w-3" fill="currentColor" />
   385→          </div>
   386→          <div
   387→            className="float-strong absolute top-1/2 -left-3 text-neon-cyan"
   388→            style={{ animationDelay: "1s" }}
   389→          >
   390→            <Sparkles className="h-3 w-3" fill="currentColor" />
   391→          </div>
   392→
   393→          {/* Heart icon — pulsing with love */}
   394→          <div
   395→            className="fade-up relative flex h-full w-full items-center justify-center"
   396→          >
   397→            <div
   398→              className="pulse-soft flex h-20 w-20 items-center justify-center rounded-full"
   399→              style={{
   400→                background:
   401→                  "linear-gradient(135deg, #C25B7E 0%, #B5453A 50%, #C25B7E 100%)",
   402→                boxShadow:
   403→                  "0 4px 20px -2px rgba(194, 91, 126, 0.50), 0 0 0 4px rgba(255, 255, 255, 0.80), 0 0 0 5px rgba(194, 91, 126, 0.20)",
   404→              }}
   405→            >
   406→              <div className="pulse-soft">
   407→                <Heart
   408→                  className="h-9 w-9 text-white"
   409→                  fill="white"
   410→                  strokeWidth={0}
   411→                />
   412→              </div>
   413→            </div>
   414→          </div>
   415→        </div>
   416→
   417→        {/* Thank you heading — animated blue-black gradient */}
   418→        <h3
   419→          className="fade-up font-arabic text-2xl font-bold sm:text-3xl"
   420→          style={{ animationDelay: "0.3s" }}
   421→        >
   422→          <span className="text-blue-black-animated">شكراً لك من كلّ قلبنا! ❤</span>
   423→        </h3>
   424→
   425→        {/* Warm message */}
   426→        <p
   427→          className="fade-up mt-3 text-[14px] leading-loose text-gray"
   428→          style={{ animationDelay: "0.5s" }}
   429→        >
   430→          تمّ استلام طلبك بنجاح. نحن سعداء بثقتك بنا، وسنتواصل معك قريباً
   431→          لتأكيد الطلب وترتيب التوصيل.
   432→        </p>
   433→
   434→        {/* Order reference — elegant badge with animated black */}
   435→        <div
   436→          className="fade-up mt-4 inline-flex items-center gap-2 rounded-full border border-charcoal/25 bg-white px-4 py-2"
   437→          style={{
   438→            boxShadow: "0 2px 12px -2px rgba(15, 13, 10, 0.18)",
   439→            animationDelay: "0.7s",
   440→          }}
   441→        >
   442→          <span className="text-[11px] text-gray">رقم الطلب</span>
   443→          <span className="font-arabic text-sm font-bold text-animated-black">
   444→            {orderRef}
   445→          </span>
   446→        </div>
   447→
   448→        {/* Order summary — elegant card with soft glow */}
   449→        <div
   450→          className="fade-up mt-5 overflow-hidden rounded-2xl border border-clay/40 bg-white p-4 text-right text-[12px]"
   451→          style={{
   452→            boxShadow:
   453→              "0 4px 20px -6px rgba(74, 85, 104, 0.18), 0 0 0 1px rgba(74, 85, 104, 0.06)",
   454→            animationDelay: "0.9s",
   455→          }}
   456→        >
   457→          {/* Items */}
   458→          {orderSummary.items.map((it, i) => (
   459→            <div key={i} className="mb-1.5 flex justify-between">
   460→              <span className="text-gray line-clamp-1">{it.name}</span>
   461→              <span className="font-bold text-charcoal">×{it.quantity}</span>
   462→            </div>
   463→          ))}
   464→          <div className="my-2.5 h-px bg-clay/30" />
   465→          {/* Totals */}
   466→          <div className="mb-1 flex justify-between">
   467→            <span className="text-gray">المجموع</span>
   468→            <span className="font-medium text-charcoal">{formatPrice(orderSummary.productTotal)}</span>
   469→          </div>
   470→          {orderSummary.discountAmount > 0 && (
   471→            <div className="mb-1 flex justify-between">
   472→              <span className="text-neon-magenta">الخصم</span>
   473→              <span className="font-medium text-neon-magenta">-{formatPrice(orderSummary.discountAmount)}</span>
   474→            </div>
   475→          )}
   476→          <div className="mb-1.5 flex justify-between">
   477→            <span className="text-gray">التوصيل</span>
   478→            <span className="font-medium text-charcoal">
   479→              {orderSummary.freeShippingApplied
   480→                ? "0 دج (مجاني)"
   481→                : formatPrice(orderSummary.shippingPrice)}
   482→            </span>
   483→          </div>
   484→          <div className="mb-2.5 flex justify-between border-t border-clay/30 pt-2 text-[14px] font-bold">
   485→            <span className="text-charcoal">المجموع الكلي</span>
   486→            <span className="text-emerald">{formatPrice(orderSummary.grandTotal)}</span>
   487→          </div>
   488→          <div className="my-2.5 h-px bg-clay/30" />
   489→          {/* Customer info */}
   490→          <div className="mb-1 flex justify-between">
   491→            <span className="text-gray">الاسم</span>
   492→            <span className="font-medium text-charcoal">{orderSummary.fullName}</span>
   493→          </div>
   494→          <div className="mb-1 flex justify-between">
   495→            <span className="text-gray">الهاتف</span>
   496→            <bdi dir="ltr" className="font-medium text-charcoal">{orderSummary.phone}</bdi>
   497→          </div>
   498→          <div className="mb-1 flex justify-between">
   499→            <span className="text-gray">الولاية</span>
   500→            <span className="font-medium text-charcoal">{orderSummary.wilayaLabel}</span>
   501→          </div>
   502→          <div className="mb-1 flex justify-between">
   503→            <span className="text-gray">التوصيل</span>
   504→            <span className="font-medium text-charcoal">{orderSummary.deliveryLabel}</span>
   505→          </div>
   506→          <div className="flex justify-between">
   507→            <span className="text-gray">التاريخ</span>
   508→            <span className="text-[11px] text-charcoal">{orderSummary.date}</span>
   509→          </div>
   510→        </div>
   511→
   512→        {/* "With love" footer */}
   513→        <p
   514→          className="fade-up mt-4 flex items-center justify-center gap-1.5 text-[12px] text-gray-light"
   515→          style={{ animationDelay: "1.2s" }}
   516→        >
   517→          <span>صُنع بحبّ في</span>
   518→          <span className="font-bold text-blue-black-animated">{BRAND.name}</span>
   519→          <span
   520→            className="pulse-soft text-neon-magenta"
   521→          >
   522→            ❤
   523→          </span>
   524→        </p>
   525→
   526→        {/* Action buttons */}
   527→        <div
   528→          className="fade-up mt-5 flex flex-col gap-2"
   529→          style={{ animationDelay: "1.4s" }}
   530→        >
   531→          <button
   532→            type="button"
   533→            onClick={() => onContinueShopping?.()}
