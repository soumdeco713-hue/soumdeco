    85→function isAuthorized_(e) {
    86→  var token = getAdminToken_();
    87→  if (!token) return true; // ⚠️ If no token set, allow all (backwards compat)
    88→  // Check X-Admin-Token header (set by /api/admin route)
    89→  var headers = (e && e.headers) || {};
    90→  var provided = headers['X-Admin-Token'] || headers['x-admin-token'] || '';
    91→  return provided === token;
    92→}

    94→function doGet(e) {
    95→  e = e || {}; var p = e.parameter || {};
    96→  var action = String(p.action || '').toLowerCase();
    97→  try {
    98→    // ─── PUBLIC OPERATIONS (no auth required) ──────────────
    99→    if (action === 'stock') return serveStock();
   100→    if (action === 'products') return serveProducts();
   101→    if (action === 'order') return doCreateOrderFromParams(p);
   102→    if (action === 'health') return jsonOut({ ok: true, time: new Date().toISOString(), sheet: SpreadsheetApp.getActiveSpreadsheet().getName() });
   103→    if (action === 'statistics') return setupStatistics();
   104→    // process_confirmed is called from the sheet menu (no HTTP) — keep public
   105→    if (action === 'process_confirmed') return doProcessConfirmedFromUrl();
   106→
   107→    // ─── PROTECTED OPERATIONS (require admin token) ───────
   108→    if (action === 'product_delete' || action === 'product_reset' ||
   109→        action === 'dedupe' || action === 'cleanup') {
   110→      if (!isAuthorized_(e)) {
   111→        return jsonOut({ ok: false, error: 'unauthorized', message: 'Admin token required for write operations' });
   112→      }
   113→    }
   114→    if (action === 'product_delete') return doDeleteProduct(p.id || '');
   115→    if (action === 'product_reset') return doResetProducts();
   116→    if (action === 'dedupe') return doDedupeProducts();
   117→    if (action === 'cleanup') return doCleanupSheet();
   118→
   119→    return jsonOut({ ok: false, error: 'unknown action: ' + action });
   120→  } catch (err) { return jsonOut({ ok: false, error: String(err) }); }
   121→}

   277→/**
   278→ * extractVariantFromRow_ — extract variant + stockKey from a row.
   279→ * Tries FOUR sources in order:
   280→ *   1. Existing Variant column (already populated)
   281→ *   2. Product name parentheses  "(المقاس: 06L)"
   282→ *   3. Notes column patterns     "المقاس: 06L · ..." (where variant is currently being written)
   283→ *   4. Returns empty if none found
   284→ *
   285→ * Returns: { variant: string, stockKey: string }
   286→ */
   287→function extractVariantFromRow_(productName, notes, existingVariant, existingStockKey) {
   288→  var variantStr = String(existingVariant || '').trim();
   289→  var stockKeyStr = String(existingStockKey || '').trim();
   290→  if (stockKeyStr) return { variant: variantStr, stockKey: stockKeyStr }; // already have both
   291→
   292→  // Strip trailing ×N
   293→  var bareName = String(productName || '').replace(/\s*[×x]\s*\d+\s*$/, '').trim();
   294→
   295→  // ─── SOURCE 1: existing variant column ───────────────────────
   296→  if (variantStr) {
   297→    return buildStockKey_(bareName, variantStr);
   298→  }
   299→
   300→  // ─── SOURCE 2: product name parentheses ──────────────────────
   301→  var variantMatch = bareName.match(/\(([^)]+)\)\s*$/);
   302→  if (variantMatch) {
   303→    var extractedVariant = parseVariantContent_(variantMatch[1]);
   304→    if (extractedVariant) {
   305→      return buildStockKey_(bareName, extractedVariant);
   306→    }
   307→  }
   308→
   309→  // ─── SOURCE 3: notes column (current frontend writes variant here) ──
   310→  // Notes format: "اللون: Blue · المقاس: Large · [user notes] · [company]"
   311→  // Or just: "المقاس: 06L"
   312→  if (notes) {
   313→    var notesVariant = extractVariantFromNotes_(notes);
   314→    if (notesVariant) {
   315→      return buildStockKey_(bareName, notesVariant);
   316→    }
   317→  }
   318→
   319→  return { variant: '', stockKey: '' };
   320→}
   321→
   322→/** Parse the content inside parentheses: "اللون: Red · المقاس: Large" → "Red - Large" */
   323→function parseVariantContent_(content) {
   324→  var parts = String(content || '').split('·');
   325→  var values = [];
   326→  for (var i = 0; i < parts.length; i++) {
   327→    var part = parts[i].trim();
   328→    if (!part) continue;
   329→    var colonIdx = part.lastIndexOf(':');
   330→    if (colonIdx >= 0) {
   331→      var value = part.substring(colonIdx + 1).trim();
   332→      // Strip emojis + variation selectors + normalize whitespace
   333→      value = value
   334→        .replace(/[\u{1F000}-\u{1FFFF}]/u, '')
   335→        .replace(/[\u{2600}-\u{27BF}]/u, '')
   336→        .replace(/[\uFE0F\u200D]/g, '')
   337→        .replace(/\s+/g, ' ')
   338→        .trim();
   339→      if (value) values.push(value);
   340→    } else if (part) {
   341→      values.push(part);
   342→    }
   343→  }
   344→  return values.join(' - ');
   345→}

   347→/** Extract variant from notes — looks for "Label: value" patterns. */
   348→function extractVariantFromNotes_(notes) {
   349→  var notesStr = String(notes || '').trim();
   350→  if (!notesStr) return '';
   351→
   352→  // Common Arabic + French variant labels (lowercased comparison)
   353→  var labels = [
   354→    'المقاس', 'اللون', 'الحجم', 'الوزن', 'النوع', 'النموذج',
   355→    'Taille', 'Couleur', 'Modèle', 'Size', 'Color'
   356→  ];
   357→
   358→  var parts = notesStr.split('·');
   359→  var values = [];
   360→  for (var i = 0; i < parts.length; i++) {
   361→    var part = parts[i].trim();
   362→    if (!part) continue;
   363→    var colonIdx = part.lastIndexOf(':');
   364→    if (colonIdx < 0) continue;
   365→    var label = part.substring(0, colonIdx).trim().toLowerCase();
   366→    var value = part.substring(colonIdx + 1).trim();
   367→    if (!value) continue;
   368→
   369→    var isVariant = false;
   370→    for (var l = 0; l < labels.length; l++) {
   371→      if (label === labels[l].toLowerCase()) { isVariant = true; break; }
   372→    }
   373→
   374→    if (isVariant) {
   375→      // Strip emojis + variation selectors (U+FE0F) + zero-width joiners (U+200D)
   376→      // and normalize whitespace. This handles cases like "06L 🚚" → "06L"
   377→      // and "Red ❤️" → "Red" (the heart has a variation selector).
   378→      value = value
   379→        .replace(/[\u{1F000}-\u{1FFFF}]/u, '')
   380→        .replace(/[\u{2600}-\u{27BF}]/u, '')
   381→        .replace(/[\uFE0F\u200D]/g, '')
   382→        .replace(/\s+/g, ' ')
   383→        .trim();
   384→      if (value) values.push(value);
   385→    }
   386→  }
   387→  return values.join(' - ');
   388→}

   390→/** Build stockKey from cleanProductName + variantStr.
   391→ *  "Cocotte...Ref 01" + "06L" → "Cocotte...Ref 01 - 06L"
   392→ *  "Cocotte...Ref 01" + "Red - Large" → "Cocotte...Ref 01 - Red;Cocotte...Ref 01 - Large"
   393→ *
   394→ *  IMPORTANT: Uses ";" as separator (NOT ",") because product names can contain
   395→ *  commas (e.g. "Cocotte minute 06, 08, 10, 12 litres Ref 01"). Using ";" ensures
   396→ *  we can reliably split the stockKey back into individual keys later.
   397→ */
   398→function buildStockKey_(bareName, variantStr) {
   399→  var cleanProductName = String(bareName || '').replace(/\s*\([^)]+\)\s*$/, '').trim();
   400→  var variantValues = String(variantStr || '').split(' - ');
   401→  var keys = [];
   402→  for (var k = 0; k < variantValues.length; k++) {
   403→    var v = variantValues[k].trim();
   404→    if (v) keys.push(cleanProductName + ' - ' + v);
   405→  }
   406→  return { variant: variantStr, stockKey: keys.join(';') };
   407→}
   408→
   409→/** Split stockKey into individual keys.
   410→ *  ONLY splits on ";" (the new separator that handles product names with commas).
   411→ *
   412→ *  NEVER splits on "," — because product names can contain commas
   413→ *  (e.g. "Cocotte minute 06, 08, 10, 12 litres Ref 01"). Splitting on ","
   414→ *  would break a single valid key into fragments.
   415→ *
   416→ *  Legacy comma-format stockKeys (from before the semicolon fix) are
   417→ *  treated as a SINGLE key here. If they were multi-key comma-format,
   418→ *  they were already broken (the comma-in-product-name bug) and won't
   419→ *  match — admin should run "Process Pending Confirmed Orders" to
   420→ *  re-extract variants with the new semicolon format.
   421→ */
   422→function splitStockKey_(stockKeyStr) {
   423→  if (!stockKeyStr) return [];
   424→  var s = String(stockKeyStr).trim();
   425→  if (!s) return [];
   426→  // Only split on ";" — never on ","
   427→  if (s.indexOf(';') >= 0) return s.split(';');
   428→  return [s]; // single key (even if it contains commas)
   429→}

   431→/**
   432→ * ============================================================
   433→ *  setupTriggers — installs the onEdit trigger AUTOMATICALLY.
   434→ *  Run this ONCE from the Apps Script editor or the spreadsheet menu.
   435→ *  After running, all status changes to "Confirmed" / "Cancelled"
   436→ *  will auto-decrement / revert stock with no further setup.
   437→ *
   438→ *  Re-running is safe — duplicates are detected + removed.
   439→ * ============================================================

   441→function setupTriggers() {
   442→  var ss = SpreadsheetApp.getActiveSpreadsheet();
   443→  var triggers = ScriptApp.getProjectTriggers();
   444→  var existing = 0;
   445→  for (var i = 0; i < triggers.length; i++) {
   446→    var t = triggers[i];
   447→    if (t.getHandlerFunction() === 'onStockEdit') {
   448→      existing++;
   449→      // Remove duplicates (keep only the first one)
   450→      if (existing > 1) {
   451→        ScriptApp.deleteTrigger(t);
   452→      }
   453→    }
   454→  }
   455→  if (existing === 0) {
   456→    ScriptApp.newTrigger('onStockEdit')
   457→      .forSpreadsheet(ss)
   458→      .onEdit()
   459→      .create();
   460→  }
   461→  // Also ensure sheets exist with the right headers
   462→  setupAllSheets();
   463→  ss.toast('✅ Auto-stock trigger installed. Confirmed orders will now decrement variant stock automatically.', 'Setup Complete', 10);
   464→}

   627→    variantsBackfilled: variantUpdates.length,
