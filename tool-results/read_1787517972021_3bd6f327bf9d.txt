     6→export type Variation = {
     7→  /** Variation name — e.g. "اللون", "الحجم", "السعة" */
     8→  name: string;
     9→  /** Options — e.g. ["أبيض", "أسود", "ذهبي"] */
    10→  options: string[];
    11→};
    12→
    13→/** Simpler per-variant model — a single color, size, or custom variable
    14→ * entry with optional price adjustment. */
    15→export type ProductVariant = {
    16→  /** Variant type: "color", "size", or any custom string (e.g. "weight", "material") */
    17→  type: string;
    18→  name: string;
    19→  /** Added to the base price (can be 0 or negative). Optional. */
    20→  priceAdjustment?: number;
    21→};
    22→
    23→/** Quantity tier — a special offer triggered by the customer's order quantity.
    24→ * A tier can combine a free-shipping benefit AND a discount amount (or just one, or both).
    25→ *
    26→ * Two modes:
    27→ *  - `mode: "exact"`   → triggers ONLY at exactly `qty` (e.g. "buy exactly 2 → free desk shipping")
    28→ *  - `mode: "min"`     → triggers at `qty` OR MORE (e.g. "buy 2+ → free shipping")
    29→ *
    30→ * The `mode` field is optional — if absent, the tier is treated as "exact" (backward compat).
    31→ * Sheet encoding: `qty:freeShipping:discountAmount:mode`
    32→ *   e.g. `2:both:0:min` = buy 2 or more → free shipping (both) + no discount
    33→ *        `3:none:500:exact` = buy exactly 3 → 500 DA discount, no free shipping
    34→ *        `2:desk:0` = buy exactly 2 → free desk shipping (legacy format, no mode = "exact")
    35→ */
    36→export type QuantityTier = {
    37→  /** The quantity that triggers this tier (e.g. 2, 3, 4) */
    38→  qty: number;
    39→  /** Whether the tier triggers at exactly `qty` ("exact") or `qty` and above ("min"). */
    40→  mode?: "exact" | "min";
    41→  /** Free-shipping benefit. "none" = no free shipping, just discount. */
    42→  freeShipping: "none" | "desk" | "home" | "both";
    43→  /** Optional discount amount in DA (0 / undefined = no discount) */
    44→  discountAmount?: number;
    45→};
    46→
    47→/** Old benefit values used by the previous tier format. Kept for backward-compat parsing. */
    48→type LegacyTierBenefit = "free_desk" | "free_home" | "free_both" | "discount";
    49→
    50→/** Map an old-format tier benefit to the new {freeShipping, discountAmount} shape. */
    51→function migrateLegacyBenefit(
    52→  benefit: LegacyTierBenefit,
    53→  discountAmount?: number,
    54→): { freeShipping: QuantityTier["freeShipping"]; discountAmount?: number } {
    55→  switch (benefit) {
    56→    case "free_desk":
    57→      return { freeShipping: "desk" };
    58→    case "free_home":
    59→      return { freeShipping: "home" };
    60→    case "free_both":
    61→      return { freeShipping: "both" };
    62→    case "discount":
    63→    default:
    64→      return { freeShipping: "none", discountAmount };
    65→  }
    66→}

   665→/**
   666→ * Split the `images` string from the sheet into an array of image URLs.
   667→ * Handles triple-tilde (new), triple-pipe (legacy), and comma-separated formats.
   668→ */
   669→function splitImageStrings(s: string): string[] {
   670→  if (!s) return [];
   671→  const trimmed = s.trim();
   672→  if (!trimmed) return [];
   673→
   674→  if (trimmed.includes("~~~")) {
   675→    return trimmed.split("~~~").map((x) => x.trim()).filter((x) => x !== "");
   676→  }
   677→  if (trimmed.includes("|||")) {
   678→    return trimmed.split("|||").map((x) => x.trim()).filter((x) => x !== "");
   679→  }
   680→  if (trimmed.includes("|")) {
   681→    return trimmed.split("|").map((x) => x.trim()).filter((x) => x !== "");
   682→  }
   683→  if (trimmed.includes("data:")) {
   684→    return trimmed.split(/,(?=data:)/).map((x) => x.trim()).filter((x) => x !== "");
   685→  }
   686→  if (trimmed.includes(",")) {
   687→    return trimmed.split(",").map((x) => x.trim()).filter((x) => x !== "");
   688→  }
   689→  return [trimmed];
   690→}

   729→/** Parse the `variants` string from the sheet into a `ProductVariant[]`.
   730→ * Format: `type:name:priceAdjustment,type:name:priceAdjustment` (comma-separated).
   731→ * The priceAdjustment is optional (defaults to 0). Type must be "color" or "size".
   732→ * Names containing commas are NOT supported (would need a different separator). */
   733→export function parseVariants(s: string | undefined | null): ProductVariant[] {
   734→  if (!s || typeof s !== "string") return [];
   735→  const trimmed = s.trim();
   736→  if (!trimmed) return [];
   737→  const out: ProductVariant[] = [];
   738→  for (const part of trimmed.split(VARIANT_SEP)) {
   739→    const clean = part.trim();
   740→    if (!clean) continue;
   741→    // Split from the LEFT into at most 3 parts: type, name, priceAdjustment.
   742→    // Names CAN contain colons (rare but possible); the priceAdjustment is the
   743→    // LAST segment.
   744→    const firstColon = clean.indexOf(":");
   745→    if (firstColon < 0) continue;
   746→    const type = clean.slice(0, firstColon).trim();
   747→    if (!type) continue; // accept any non-empty type (color, size, or custom)
   748→    const rest = clean.slice(firstColon + 1);
   749→    const lastColon = rest.lastIndexOf(":");
   750→    let name: string;
   751→    let adjustment = 0;
   752→    if (lastColon < 0) {
   753→      name = rest.trim();
   754→    } else {
   755→      const maybeNum = Number(rest.slice(lastColon + 1));
   756→      if (isNaN(maybeNum)) {
   757→        name = rest.trim();
   758→      } else {
   759→        name = rest.slice(0, lastColon).trim();
   760→        adjustment = maybeNum;
   761→      }
   762→    }
   763→    if (!name) continue;
   764→    out.push({
   765→      type: type as "color" | "size",
   766→      name,
   767→      priceAdjustment: adjustment,
   768→    });
   769→  }
   770→  return out;
   771→}

   824→/** Parse the quantityTiers string from the sheet into an array of QuantityTier objects.
   825→ * New format with mode: "2:both:0:min,3:desk:500:exact" (qty:freeShipping:discountAmount:mode)
   826→ * New format without mode: "2:both:200,3:desk:0" (mode defaults to "exact")
   827→ * Legacy format (still supported for back-compat with existing sheet rows):
   828→ *   "2:free_both:0,3:discount:200" (qty:benefit:discountAmount) */
   829→export function parseQuantityTiers(s: string | undefined | null): QuantityTier[] {
   830→  if (!s || typeof s !== "string") return [];
   831→  const trimmed = s.trim();
   832→  if (!trimmed) return [];
   833→  const parts = trimmed.split(",").map((x) => x.trim()).filter(Boolean);
   834→  const out: QuantityTier[] = [];
   835→  const newShippingTokens: QuantityTier["freeShipping"][] = ["none", "desk", "home", "both"];
   836→  const legacyBenefits: LegacyTierBenefit[] = ["free_desk", "free_home", "free_both", "discount"];
   837→  const modeTokens: QuantityTier["mode"][] = ["exact", "min"];
   838→  for (const part of parts) {
   839→    const [qtyStr, middle, discountStr, modeStr] = part.split(":");
   840→    const qty = Number(qtyStr);
   841→    if (isNaN(qty) || qty < 1) continue;
   842→    const discount = Number(discountStr) || 0;
   843→    const mode: QuantityTier["mode"] =
   844→      modeStr === "min" ? "min" : modeStr === "exact" ? "exact" : "exact";
   845→    if (newShippingTokens.includes(middle as QuantityTier["freeShipping"])) {
   846→      out.push({
   847→        qty,
   848→        mode,
   849→        freeShipping: middle as QuantityTier["freeShipping"],
   850→        discountAmount: discount > 0 ? discount : undefined,
   851→      });
   852→    } else if (legacyBenefits.includes(middle as LegacyTierBenefit)) {
   853→      const migrated = migrateLegacyBenefit(middle as LegacyTierBenefit, discount > 0 ? discount : undefined);
   854→      out.push({
   855→        qty,
   856→        mode,
   857→        freeShipping: migrated.freeShipping,
   858→        discountAmount: migrated.discountAmount,
   859→      });
   860→    } else {
   861→      // Unknown token — default to no benefit so the tier is still preserved.
   862→      out.push({
   863→        qty,
   864→        mode,
   865→        freeShipping: "none",
   866→        discountAmount: discount > 0 ? discount : undefined,
   867→      });
   868→    }
   869→  }
   870→  return out;
   871→}
   872→
   873→/** Normalize a raw quantityTiers value (from localStorage, sheet JSON, or the API)
   874→ * into the new QuantityTier shape. Handles:
   875→ *  - arrays already in the new format ({qty, freeShipping, discountAmount?, mode?})
   876→ *  - arrays in the legacy format ({qty, benefit, discountAmount?}) → migrate
   877→ *  - string in new or legacy sheet format → parseQuantityTiers
   878→ */
   879→export function normalizeTiers(raw: unknown): QuantityTier[] {
   880→  if (Array.isArray(raw)) {
   881→    const newShippingTokens: QuantityTier["freeShipping"][] = ["none", "desk", "home", "both"];
   882→    const legacyBenefits: LegacyTierBenefit[] = ["free_desk", "free_home", "free_both", "discount"];
   883→    return (raw as any[])
   884→      .filter((t) => t && typeof t.qty === "number")
   885→      .map((t) => {
   886→        const qty = Number(t.qty);
   887→        const discount = t.discountAmount == null ? undefined : Number(t.discountAmount);
   888→        const mode: QuantityTier["mode"] =
   889→          t.mode === "min" ? "min" : t.mode === "exact" ? "exact" : "exact";
   890→        if (typeof t.freeShipping === "string" && newShippingTokens.includes(t.freeShipping)) {
   891→          return { qty, mode, freeShipping: t.freeShipping, discountAmount: discount } as QuantityTier;
   892→        }
   893→        if (typeof t.benefit === "string" && legacyBenefits.includes(t.benefit)) {
   894→          const migrated = migrateLegacyBenefit(t.benefit as LegacyTierBenefit, discount);
   895→          return { qty, mode, ...migrated } as QuantityTier;
   896→        }
   897→        // Unknown shape — preserve qty + discount, no shipping benefit.
   898→        return { qty, mode, freeShipping: "none" as const, discountAmount: discount } as QuantityTier;
   899→      });
   900→  }
   901→  if (typeof raw === "string" && raw.trim()) {
   902→    return parseQuantityTiers(raw);
   903→  }
   904→  return [];
   905→}

   952→function normalizeProduct(p: any): Product | null {
   953→  if (!p || typeof p !== "object") return null;
   954→  const toStr = (v: any): string => {
   955→    if (v == null) return "";
   956→    if (typeof v === "string") return v;
   957→    if (typeof v === "object" && "fr" in v) return String(v.fr ?? "");
   958→    if (typeof v === "object" && "ar" in v) return String(v.ar ?? "");
   959→    return String(v);
   960→  };
   961→  const image = toStr(p.image);
   962→  let images: string[] = [];
   963→  if (Array.isArray(p.images)) {
   964→    images = p.images.map(toStr).filter((s) => s.trim() !== "");
   965→  } else if (typeof p.images === "string" && p.images.trim() !== "") {
   966→    images = splitImageStrings(p.images);
   967→  }
   968→  images = Array.from(new Set(images));
   969→  if (image && images[0] !== image) {
   970→    images = [image, ...images.filter((i) => i !== image)];
   971→  }
   972→
   973→  let variations: Variation[] = [];
   974→  if (Array.isArray(p.variations)) {
   975→    variations = p.variations
   976→      .filter((v: any) => v && typeof v === "object" && v.name)
   977→      .map((v: any) => ({
   978→        name: String(v.name),
   979→        options: Array.isArray(v.options)
   980→          ? v.options.map((o: any) => String(o)).filter(Boolean)
   981→          : [],
   982→      }))
   983→      .filter((v: Variation) => v.options.length > 0);
   984→  } else if (typeof p.variations === "string" && p.variations.trim()) {
   985→    variations = parseVariations(p.variations);
   986→  }
   987→
   988→  const variants = normalizeVariants(p.variants);
   989→
   990→  let highlights: string[] = [];
   991→  if (Array.isArray(p.highlights)) {
   992→    highlights = p.highlights.map((h: any) => String(h)).filter((h) => h.trim() !== "");
   993→  } else if (typeof p.highlights === "string" && p.highlights.trim()) {
   994→    highlights = parseHighlights(p.highlights);
   995→  }
   996→
   997→  return {
   998→    id: String(p.id ?? ""),
   999→    name: toStr(p.name),
  1000→    description: toStr(p.description ?? ""),
  1001→    category: toStr(p.category),
  1002→    price: (() => {
  1003→      if (p.price === null || p.price === undefined) return null;
  1004→      if (typeof p.price === "object" && p.price !== null) return null;
  1005→      const n = Number(p.price);
  1006→      // P0 FIX #4: Guard against NaN — treat as null (price-on-request)
  1007→      return isNaN(n) ? null : n;
  1008→    })(),
  1009→    oldPrice: (() => {
  1010→      if (p.oldPrice === null || p.oldPrice === undefined || p.oldPrice === "") return null;
  1011→      if (typeof p.oldPrice === "object" && p.oldPrice !== null) return null;
  1012→      const n = Number(p.oldPrice);
  1013→      // P0 FIX #4: Guard against NaN
  1014→      return isNaN(n) ? null : n;
  1015→    })(),
  1016→    image,
  1017→    images,
  1018→    variations,
  1019→    variants,
  1020→    stock: (() => {
  1021→      if (p.stock === null || p.stock === undefined || p.stock === "") return null;
  1022→      if (typeof p.stock === "object" && p.stock !== null) return null;
  1023→      const n = Number(p.stock);
  1024→      // P0 FIX #4: Guard against NaN
  1025→      return isNaN(n) ? null : n;
  1026→    })(),
  1027→    highlights,
  1028→    sortOrder:
  1029→      p.sortOrder === null || p.sortOrder === undefined
  1030→        ? 999
  1031→        : Number(p.sortOrder),
  1032→    badge:
  1033→      p.badge === null || p.badge === undefined
  1034→        ? ""
  1035→        : String(p.badge),
  1036→    quantityTiers: normalizeTiers(p.quantityTiers),
  1037→    featured: (p.featured === true ||
  1038→               p.featured === 1 ||
  1039→               p.featured === "1" ||
  1040→               (typeof p.featured === "string" &&
  1041→                p.featured.toLowerCase() === "true")),
  1042→    inStock: p.inStock !== false, // default true unless explicitly false
  1043→    isSpecialOffer: (p.isSpecialOffer === true ||
  1044→                     p.isSpecialOffer === 1 ||
  1045→                     p.isSpecialOffer === "1" ||
  1046→                     (typeof p.isSpecialOffer === "string" &&
  1047→                      p.isSpecialOffer.toLowerCase() === "true")),
  1048→  };
  1049→}

  1051→export function saveCatalog(products: Product[]): boolean {
  1052→  if (typeof window === "undefined") return false;
  1053→  const json = JSON.stringify(products);
  1054→  try {
  1055→    window.localStorage.setItem(CATALOG_STORAGE_KEY, json);
  1056→    // Verify it was actually saved (some browsers silently truncate)
  1057→    const check = window.localStorage.getItem(CATALOG_STORAGE_KEY);
  1058→    if (check && check.length === json.length) {
  1059→      return true;
  1060→    }
  1061→    // Verification failed — clear and fall through to IndexedDB
  1062→    window.localStorage.removeItem(CATALOG_STORAGE_KEY);
  1063→    console.warn("[saveCatalog] localStorage write verification failed, falling back to IndexedDB");
  1064→  } catch (e) {
  1065→    // localStorage quota exceeded — likely a large catalog.
  1066→    console.warn("[saveCatalog] localStorage quota exceeded, falling back to IndexedDB");
  1067→    try {
  1068→      window.localStorage.removeItem(CATALOG_STORAGE_KEY);
  1069→    } catch {}
  1070→  }
  1071→  // Fall back to IndexedDB (async, but we can't await here).
  1072→  // Fire-and-forget — callers that need guaranteed save should use saveCatalogAsync()
  1073→  import("./adaptive-storage")
  1074→    .then(({ adaptiveSet }) => adaptiveSet(CATALOG_STORAGE_KEY, json))
  1075→    .catch(() => {});
  1076→  return false; // localStorage failed — return false so callers know
  1077→}
